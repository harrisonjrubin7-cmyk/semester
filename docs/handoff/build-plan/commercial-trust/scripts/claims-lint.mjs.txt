// Claims linter. node claims-lint.mjs <files...> — flags prohibited claims that aren't negated nearby.
import { readFileSync } from "node:fs";
const RX = /FERPA[- ]compliant|\bcertified\b|all-in-one|AI-powered|system of record|replaces? your|guaranteed|\bsecure\b|bank-grade|military-grade/gi;
const NEG = /\b(not|never|no|isn['’]t|aren['’]t|doesn['’]t|don['’]t|without|stays?|remains?|until|before|only when|instead of|rather than)\b/i;
let bad = 0;
for (const f of process.argv.slice(2)) {
  const t = readFileSync(f, "utf8").replace(/<[^>]+>/g, " ").replace(/\s+/g, " ");
  for (const m of t.matchAll(RX)) {
    const before = t.slice(Math.max(0, m.index - 60), m.index);
    if (!NEG.test(before)) { bad++; console.log(`REVIEW ${f}: “…${t.slice(Math.max(0, m.index - 80), m.index + 60)}…”`); }
  }
}
process.exit(bad ? 1 : 0);
