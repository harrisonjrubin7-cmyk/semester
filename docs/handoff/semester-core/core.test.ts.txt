import { describe, it, expect } from 'vitest';
import { resolve, effectiveOrigin, ageLabel } from './provenance';
import { order } from './nextaction';
import { canRead, level } from './access';
import { state, optimistic, POLICY } from './freshness';
import { pattern } from './consequence';
import { gate } from './aigateway';
import { step, Approval } from './approvals';
import { weighted } from './scoring';
import { readiness } from './pilot';
const NOW = Date.UTC(2026, 9, 7, 12); const H = 3600e3;

describe('provenance', () => {
  it('official beats personal; stale official loses to personal', () => {
    expect(resolve([{ value: 'Thu', origin: 'personal', updatedAt: NOW }, { value: 'Wed', origin: 'official', updatedAt: NOW - H }], NOW)!.shown.value).toBe('Wed');
    expect(effectiveOrigin({ value: 1, origin: 'connected', updatedAt: NOW - 7 * H }, NOW)).toBe('stale');
    const r = resolve([{ value: 'Thu', origin: 'personal', updatedAt: NOW }, { value: 'Wed', origin: 'connected', updatedAt: NOW - 7 * H }], NOW)!;
    expect(r.shown.value).toBe('Thu'); expect(r.conflict).toBe(true);
  });
  it('never invents a time', () => { expect(ageLabel(undefined, NOW)).toBe('Update time not recorded'); expect(ageLabel(NOW - 3 * 86400e3, NOW)).toBe('Updated 3 days ago'); });
});
describe('next action', () => {
  it('a blocking hold outranks a low-weight task due sooner', () => {
    const r = order([{ id: 't', title: 'Reading', kind: 'task', origin: 'official', dueAt: NOW + 20 * H, weight: 0.02 }, { id: 'h', title: 'Hold', kind: 'hold', origin: 'official', dueAt: NOW + 5 * 24 * H, blocks: ['registration'] }], NOW);
    expect(r[0].id).toBe('h'); expect(r[0].reasons).toContain('Blocks registration');
  });
  it('drops done and snoozed', () => { expect(order([{ id: 'a', title: '', kind: 'task', origin: 'personal', done: true }, { id: 'b', title: '', kind: 'task', origin: 'personal', snoozedUntil: NOW + H }], NOW)).toEqual([]); });
});
describe('access', () => {
  it('guardian needs an unexpired, unrevoked share', () => {
    const c = { role: 'guardian' as const, userId: 'g', now: NOW };
    expect(canRead('grades', c).ok).toBe(false);
    expect(canRead('grades', { ...c, shares: [{ grantee: 'g', screen: 'grades', expiresAt: NOW + H }] }).ok).toBe(true);
    expect(canRead('grades', { ...c, shares: [{ grantee: 'g', screen: 'grades', expiresAt: NOW - 1 }] }).ok).toBe(false);
  });
  it('break-glass needs two other approvers', () => {
    const c = { role: 'security' as const, userId: 's', now: NOW };
    expect(canRead('privacy', { ...c, grants: [{ kind: 'breakglass' as const, id: 'b', expiresAt: NOW + H, approvedBy: ['s', 'x'] }] }).ok).toBe(false);
    expect(canRead('privacy', { ...c, grants: [{ kind: 'breakglass' as const, id: 'b', expiresAt: NOW + H, approvedBy: ['x', 'y'] }] }).ok).toBe(true);
  });
  it('unknown role/screen is none', () => { expect(level('partner', 'grades')).toBe('none'); });
});
describe('freshness', () => {
  it('maps lag to policy', () => {
    expect(state({ lastEventAt: NOW, appliedAt: NOW - 3000 }, NOW)).toBe('live');
    expect(POLICY[state({ lastEventAt: NOW, appliedAt: NOW - 120e3 }, NOW)].destructive).toBe('confirm-at-source');
    expect(POLICY[state({ lastEventAt: NOW, workerFailed: true }, NOW)].destructive).toBe('disabled');
  });
  it('optimistic confirms on version, rolls back on timeout', () => { expect(optimistic(5, 5, NOW, NOW)).toBe('confirmed'); expect(optimistic(5, 4, NOW - 20e3, NOW)).toBe('rolled-back'); });
});
describe('consequence', () => { it('picks the pattern', () => { expect(pattern({ reversible: true })).toBe('undo'); expect(pattern({ reversible: true, leavesSemester: true })).toBe('preview'); expect(pattern({ reversible: false, record: true })).toBe('review-confirm'); expect(pattern({ reversible: false, institutional: true })).toBe('two-person'); }); });
describe('AI gateway', () => {
  const policy = { courseAllows: ['outline' as const], institutionAiOn: true };
  it('never sends student records', () => { const d = gate({ task: 'outline', scope: [{ id: 'syl', cls: 'course' }, { id: 'tx', cls: 'student-record' }], policy, budgetLeft: 3 }); expect(d.allowed).toBe(true); expect(d.sendScope).toEqual(['syl']); expect(d.dropped).toEqual(['tx']); });
  it('enforces course policy', () => { expect(gate({ task: 'write', scope: [], policy, budgetLeft: 3 }).allowed).toBe(false); });
});
describe('approvals', () => {
  const a: Approval = { id: '1', requester: 'r', approvers: [], state: 'requested', expiresAt: NOW + H, mfaAt: { x: NOW, y: NOW, r: NOW } };
  it('two distinct non-requesters with fresh MFA', () => {
    expect('error' in step(a, { type: 'approve', by: 'r', at: NOW })).toBe(true);
    const one = step(a, { type: 'approve', by: 'x', at: NOW }) as Approval; expect(one.state).toBe('approved-1');
    expect('error' in step(one, { type: 'approve', by: 'x', at: NOW })).toBe(true);
    const two = step(one, { type: 'approve', by: 'y', at: NOW }) as Approval; expect(two.state).toBe('approved');
    expect((step(two, { type: 'execute', by: 'r', at: NOW }) as Approval).state).toBe('executed');
  });
});
describe('scoring', () => { it('renormalises over graded criteria', () => { expect(weighted({ correctness: 5, codeQuality: 5 }).of5).toBe(5); expect(weighted({}).of5).toBeNull(); expect(weighted({ correctness: 4 }).missing).toContain('rendering'); }); });
describe('pilot', () => { it('ready only when every P0 is done', () => { expect(readiness([{ priority: 'P0', status: 'Done', gate: 'x' }, { priority: 'P0', status: 'Not started', gate: 'Pilot contract' }]).blockers.length).toBe(1); }); });
