async function request(url, scope, options = {}) {
const token = await getToken([SCOPE + scope]);
const response = await fetch(
trustedUrl(url, config.serviceOrigins),
{
...options,
redirect: "error",
signal: AbortSignal.timeout(10_000),
headers: {
...options.headers,
Authorization: `Bearer ${token.accessToken}`
}
}
);
if (!response.ok) {
const error = new Error(
`AGS_HTTP_${response.status}`
);
error.status = response.status;
error.retryAfter =
response.headers.get("retry-after");
throw error;
}
return response;
}
