export function exportSequence(events) {
const participants = {
browser: "Browser",
lms: "LMS",
tool: "Semester",
store: "Launch Store",
ags: "AGS"
};
function cleanLabel(value) {
return String(value)
.replace(/[\r\n]/g, " ")
.replace(/[^a-zA-Z0-9 .,():_\/-]/g, " ")
.slice(0, 160);
}
const lines = [
"sequenceDiagram",
...Object.entries(participants).map(
([id, label]) =>
` 	participant ${id} as ${label}`
)
];
for (const event of events) {
if (
!participants[event.from] ||
!participants[event.to]
) {
throw new Error("TRACE_PARTICIPANT_INVALID");
}
const arrow = event.response ? "-->>" : "->>";
lines.push(
` 	${event.from}${arrow}${event.to}: ` +
cleanLabel(event.label)
);
}
return lines.join("\n") + "\n";
}
