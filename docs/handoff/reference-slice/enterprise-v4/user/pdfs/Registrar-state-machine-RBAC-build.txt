export class WorkflowEngine {
constructor(
private readonly repo: WorkflowRepository,
private readonly guards: GuardServices,
private readonly hooks: HookServices
) {}
async transition(
ctx: TransitionContext,
action: WorkflowAction
): Promise<WorkflowState> {
const machine = await this.repo.getMachine();
const stateDef = machine.states[ctx.state];
if (!stateDef) {
throw new TransitionError("Unknown state", "STATE_UNKNOWN");
}
const candidates = stateDef.transitions.filter(
t => t.action === action && t.actors.includes(ctx.actorRole)
);
if (candidates.length === 0) {
throw new TransitionError(
"Action not allowed for role/state",
"ACTION_NOT_ALLOWED"
);
}
let selected: TransitionDefinition | null = null;
for (const candidate of candidates) {
const ok = await this.evaluateGuards(
ctx,
candidate.guards ?? []
);
if (ok) {
selected = candidate;
break;
}
}
if (!selected) {
throw new TransitionError(
"No transition guards passed",
"GUARD_FAILED"
);
}
const toState = selected.to;
await this.repo.inTransaction(async () => {
const moved = await this.repo.updateState(
ctx.workflowInstanceId,
ctx.state,
toState
);
if (!moved) {
throw new TransitionError(
"State changed concurrently",
"CONCURRENT_MODIFICATION"
);
}
const audit: AuditInput = {
actorUserId: ctx.actorUserId,
actedAsRoleCode: ctx.actorRole,
action,
fromState: ctx.state,
toState,
workflowInstanceId: ctx.workflowInstanceId,
entityId: ctx.entityId,
scopeType: ctx.scopeType,
scopeId: ctx.scopeId,
comment: ctx.comment,
metadata: ctx.metadata,
requestId: ctx.requestId
};
await this.repo.appendTransition(audit);
if (selected.hooks?.includes("auditTransition")) {
await this.hooks.auditTransition(audit);
}
if (selected.hooks?.includes("snapshotPublication")) {
await this.hooks.snapshotPublication(ctx);
}
if (selected.hooks?.includes("queueValidation")) {
await this.hooks.queueValidation(ctx);
}
if (selected.hooks?.includes("createWorkflowTask")) {
await this.hooks.createWorkflowTask(ctx, toState);
}
if (selected.hooks?.includes("notifySubmitter")) {
await this.hooks.notifySubmitter(ctx, toState);
}
if (selected.hooks?.includes("emitPublicationEvent")) {
await this.hooks.emitPublicationEvent(ctx);
}
});
return toState;
}
private async evaluateGuards(
ctx: TransitionContext,
guardNames: string[]
): Promise<boolean> {
for (const name of guardNames) {
const fn = this.guards[name as keyof GuardServices];
if (!fn) {
throw new TransitionError(
`Unknown guard: ${name}`,
"GUARD_UNKNOWN"
);
}
const ok = await fn.call(this.guards, ctx);
if (!ok) return false;
}
return true;
}
}

Notification architecture
Example notification policies
Webhook model
Subscriber endpoint registration.
Per-subscriber secret or signing key.
HMAC signature header.
At-least-once delivery.
Idempotency key on every event.
Retry with backoff for transient failures.
Dead-letter/quarantine state for repeated failure.
Delivery attempt history with timestamps and HTTP outcomes.
Webhook architecture guidance recommends explicit event identity, validation, retries,
authentication, and observable recovery paths.
CREATE TABLE outbox_events (
id uuid PRIMARY KEY,
event_type text NOT NULL,
aggregate_type text NOT NULL,
aggregate_id uuid NOT NULL,
payload jsonb NOT NULL,
status text NOT NULL CHECK (status IN ('pending','published','failed')),
created_at timestamptz NOT NULL DEFAULT now(),
published_at timestamptz
);
CREATE TABLE webhook_subscriptions (
id uuid PRIMARY KEY,
subscriber_name text NOT NULL,
callback_url text NOT NULL,
signing_secret text NOT NULL,
enabled boolean NOT NULL DEFAULT true,
created_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE webhook_deliveries (
id uuid PRIMARY KEY,
outbox_event_id uuid NOT NULL REFERENCES outbox_events(id),
webhook_subscription_id uuid NOT NULL REFERENCES webhook_subscriptions(id),
status text NOT NULL CHECK (
status IN ('queued','sent','failed','quarantined','duplicate')
),
attempt_count integer NOT NULL DEFAULT 0,
last_http_status integer,
next_attempt_at timestamptz,
last_error text,
created_at timestamptz NOT NULL DEFAULT now(),
delivered_at timestamptz
);

Delivery schema
The safest implementation pattern is:
1. Transition engine validates state + role + guards.
2. Database transaction updates current state and appends immutable transition/audit rows.
3. Outbox event is written in the same transaction.
4. Async processors deliver notifications and webhooks.
5. FERPA access log records all sensitive view/export/override actions separately from
workflow events.
That gives Semester a registrar-grade workflow backbone with explicit approval routing,
immutable evidence, and scalable event delivery.
