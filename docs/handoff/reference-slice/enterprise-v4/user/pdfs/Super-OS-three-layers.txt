Semester should be positioned as a unified Education OS that brings student experience, school
operations, and business management into one connected platform. Current higher-ed systems
increasingly emphasize centralized student records, connected advising workflows, and
institution-wide integration across admissions, enrollment, finance, and student success.
The product can be structured as three visible layers on top of one shared data and workflow
core. Modern higher-ed platforms increasingly connect students, faculty, staff, and
administrators around one lifecycle record instead of isolated systems.

Student profile schema
[6] [1]
Profile capabilities
CREATE TABLE persons (
id uuid PRIMARY KEY,
institutional_id text UNIQUE,
lifecycle_stage text NOT NULL CHECK (
lifecycle_stage IN ('prospect','applicant','admitted','matriculating','student',
),
legal_first_name text,
legal_last_name text,
preferred_name text,
primary_email text UNIQUE NOT NULL,
created_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE profiles (
id uuid PRIMARY KEY,
person_id uuid NOT NULL UNIQUE REFERENCES persons(id),
profile_slug text UNIQUE,
headline text,
bio text,
photo_url text,
major_label text,
class_year text,
hometown text,
pronouns text,
default_visibility text NOT NULL CHECK (
default_visibility IN ('public','campus','connections','private')
),
messaging_enabled boolean NOT NULL DEFAULT true,
portfolio_enabled boolean NOT NULL DEFAULT true,
created_at timestamptz NOT NULL DEFAULT now(),
updated_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE profile_fields (
id uuid PRIMARY KEY,
profile_id uuid NOT NULL REFERENCES profiles(id),
field_key text NOT NULL,
field_value jsonb NOT NULL,
UNIQUE(profile_id, field_key)
);
CREATE TABLE profile_visibility_rules (
id uuid PRIMARY KEY,
profile_id uuid NOT NULL REFERENCES profiles(id),
field_key text NOT NULL,
audience_type text NOT NULL CHECK (
audience_type IN ('public','campus','connections','advisor','faculty','org','priv
),
audience_value text,
can_view boolean NOT NULL DEFAULT true
);
CREATE TABLE portfolio_items (
id uuid PRIMARY KEY,
Database schema
profile_id uuid NOT NULL REFERENCES profiles(id),
item_type text NOT NULL CHECK (
item_type IN ('coursework','project','credential','award','publication','presenta
),
title text NOT NULL,
subtitle text,
description text,
evidence_url text,
media_url text,
issued_by_org_id uuid,
related_course_id uuid,
visibility text NOT NULL CHECK (
visibility IN ('public','campus','connections','private')
),
verification_status text NOT NULL DEFAULT 'self_added' CHECK (
verification_status IN ('self_added','faculty_verified','institution_verified')
),
start_date date,
end_date date,
sort_order integer NOT NULL DEFAULT 0
);
CREATE TABLE credentials (
id uuid PRIMARY KEY,
person_id uuid NOT NULL REFERENCES persons(id),
credential_type text NOT NULL CHECK (
credential_type IN ('badge','certificate','microcredential','degree_progress_mile
),
title text NOT NULL,
issuer_org_id uuid,
issued_at timestamptz,
expires_at timestamptz,
metadata jsonb NOT NULL DEFAULT '{}'::jsonb
);
CREATE TABLE profile_links (
id uuid PRIMARY KEY,
profile_id uuid NOT NULL REFERENCES profiles(id),
label text NOT NULL,
url text NOT NULL,
visibility text NOT NULL CHECK (
visibility IN ('public','campus','private')
)
);
