A strong registrar workflow should move from department dra through chair/dean review,
registrar validation, Academic Affairs exceptions when needed, and final publication, with
rule-based checks for meeting patterns, room use, and overlaps. University scheduling policies
commonly require standard meeting patterns, designated departmental submissions,
chair/dean review, registrar conformity checks, and special approval for non-standard times or
exceptional offerings.
The full schedule workflow should separate ordinary approvals from exception approvals,
because non-standard meeting times, special room needs, and unusual offerings oen require
dean or Academic Affairs involvement beyond basic registrar review. Several registrar policies
explicitly require dean review and final approval for non-standard requests or special
scheduling exceptions.

The conflict engine should run at save time, submission time, nightly batch scans, and pre-
publish certification. Scheduling best practices explicitly call out the need to detect double-
booked instructors, room overlaps, and rule violations automatically.

function overlaps(a, b) {
return (
a.day === b.day &&
a.start < b.end &&
b.start < a.end &&
a.dateStart <= b.dateEnd &&
b.dateStart <= a.dateEnd
);
}
function detectInstructorConflicts(meetings) {
const byInstructor = groupBy(
meetings.filter(m => m.instructorId),
m => m.instructorId
);
const conflicts = [];
for (const instructorId of Object.keys(byInstructor)) {
const items = byInstructor[instructorId];
for (let i = 0; i < items.length; i++) {
for (let j = i + 1; j < items.length; j++) {
if (overlaps(items[i], items[j])) {
conflicts.push({
type: "INSTRUCTOR_OVERLAP",
severity: "BLOCKING",
instructorId,
leftSectionId: items[i].sectionId,
rightSectionId: items[j].sectionId
});
}
}
}
}
return conflicts;
}
