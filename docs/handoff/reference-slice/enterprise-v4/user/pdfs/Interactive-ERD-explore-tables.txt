CREATE TABLE semester.disclosure_retention_reviews (
tenant_id uuid NOT NULL,
subject_person_id uuid NOT NULL,
education_records_retained boolean NOT NULL DEFAULT true,
institutional_hold boolean NOT NULL DEFAULT true,
policy_version_id uuid NOT NULL,
retirement_evidence_reference text,
reviewed_by uuid NOT NULL,
reviewed_at timestamptz NOT NULL,
review_valid_until timestamptz NOT NULL,
PRIMARY KEY (tenant_id, subject_person_id),
FOREIGN KEY (tenant_id, subject_person_id)
REFERENCES semester.people (tenant_id, id),
FOREIGN KEY (tenant_id, policy_version_id)
REFERENCES semester.policy_versions (tenant_id, id),
FOREIGN KEY (tenant_id, reviewed_by)
REFERENCES semester.people (tenant_id, id),
CHECK (review_valid_until > reviewed_at),
CHECK (
education_records_retained
OR retirement_evidence_reference IS NOT NULL
)
);
Only an authorized records workflow may update this state. Deleting a Semester account does
not establish that the institution no longer maintains education records.
CREATE FUNCTION semester.guard_disclosure_retention()
RETURNS trigger
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = pg_catalog
AS $$
DECLARE
r semester.disclosure_retention_reviews%ROWTYPE;
BEGIN
SELECT *
INTO r
FROM semester.disclosure_retention_reviews
WHERE tenant_id = OLD.tenant_id
AND subject_person_id = OLD.subject_person_id
Retention state
Deletion guard
FOR UPDATE;
IF NOT FOUND THEN
RAISE EXCEPTION
'Disclosure deletion blocked: no approved retention review'
USING ERRCODE = '55000';
END IF;
IF r.review_valid_until <= statement_timestamp() THEN
RAISE EXCEPTION
'Disclosure deletion blocked: retention review expired'
USING ERRCODE = '55000';
END IF;
IF r.education_records_retained OR r.institutional_hold THEN
RAISE EXCEPTION
'Disclosure deletion blocked: records retained or hold active'
USING ERRCODE = '55000';
END IF;
IF EXISTS (
SELECT 1
FROM semester.student_data_requests q
WHERE q.tenant_id = OLD.tenant_id
AND q.subject_person_id = OLD.subject_person_id
AND q.request_type = 'inspect'
AND q.status NOT IN ('fulfilled', 'closed')
) THEN
RAISE EXCEPTION
'Disclosure deletion blocked: inspection request outstanding'
USING ERRCODE = '55000';
END IF;
RETURN OLD;
END;
$$;
REVOKE ALL ON FUNCTION
semester.guard_disclosure_retention()
FROM PUBLIC;
CREATE TRIGGER guard_disclosure_retention
BEFORE DELETE ON semester.ferpa_disclosure_records
FOR EACH ROW
EXECUTE FUNCTION semester.guard_disclosure_retention();
Apply default-deny RLS and reviewed grants to the retention table.
Use a dedicated disposition role with visibility into every applicable request and hold.
Do not allow ordinary users to set retirement flags.
Audit approved disposition before removing records.
Revoke TRUNCATE; row-level deletion triggers do not handle it.
Mandatory safeguards around the trigger
Coordinate request receipt, hold placement, and deletion using shared locking or an
equivalent concurrency protocol.
Block on missing, stale, or incomplete retirement evidence.
Do not treat a hidden request—filtered out by RLS—as proof that no request exists.
Maintain separate policies for operational logs, required disclosure records, files, backups,
and externally held records.
PostgreSQL explicitly excludes TRUNCATE from row-security enforcement. This is why grants and
operational controls are necessary alongside triggers.
