| "learning.practice.use"

| "learning.submission.read_own"
| "learning.submission.read_assigned"
| "learning.grade.write_assigned"
| "advising.course_progress.read_assigned";
type Purpose =
| "course_learning"
| "course_instruction"
| "academic_advising";

export type Decision = {
allow: boolean;
reason:
| "ALLOWED"
| "TENANT_MISMATCH"
| "INACTIVE_MEMBERSHIP"
| "LTI_CONTEXT_MISMATCH"
| "LTI_CAPABILITY_DENIED"
| "PERMISSION_MISSING"
| "RELATIONSHIP_MISSING"
| "PURPOSE_MISMATCH"
| "FIELD_SCOPE_DENIED";
permittedFields: readonly string[];
policyVersion: string;
};
interface AuthorizationRepository {
// Load through a trusted server connection.
// Identity comes from the verified session.
// Facts must apply to this actor, section and subject.
loadFacts(input: {
actor: Actor;
resource: CourseResource;
}): Promise<AuthorizationFacts>;
}
const PURPOSES: Record<CoursePermission, readonly Purpose[]> = {
"learning.resource.read": [
"course_learning",
"course_instruction",
],
"learning.practice.use": ["course_learning"],
"learning.roster.read_assigned": ["course_instruction"],
"learning.submission.read_own": ["course_learning"],
"learning.submission.read_assigned": ["course_instruction"],
"learning.grade.write_assigned": ["course_instruction"],
"advising.course_progress.read_assigned": [
"academic_advising",
],
};
function relationshipAllows(
permission: CoursePermission,
actor: Actor,
resource: CourseResource,
facts: AuthorizationFacts,
): boolean {
switch (permission) {
case "learning.resource.read":
return (
facts.activeLearnerEnrollment ||
facts.activeInstructorAssignment
);
case "learning.practice.use":
return facts.activeLearnerEnrollment;
case "learning.submission.read_own":
return (
facts.activeLearnerEnrollment &&
resource.subjectPersonId === actor.personId
);
case "learning.roster.read_assigned":
case "learning.submission.read_assigned":
case "learning.grade.write_assigned":
return facts.activeInstructorAssignment;
case "advising.course_progress.read_assigned":
return (
resource.subjectPersonId !== undefined &&
facts.activeAdvisorAssignment
);
}
}
export async function evaluateCoursePermission(
input: {
actor: Actor;
resource: CourseResource;
permission: CoursePermission;
purpose: Purpose;
requestedFields: readonly string[];
lti?: VerifiedLtiContext;
},
repo: AuthorizationRepository,
): Promise<Decision> {
const deny = (
reason: Decision["reason"],
policyVersion = "not-evaluated",
): Decision => ({
allow: false,
reason,
permittedFields: [],
policyVersion,
});
const { actor, resource, permission, purpose, lti } = input;
if (actor.tenantId !== resource.tenantId) {
return deny("TENANT_MISMATCH");
}
if (lti) {
if (
lti.tenantId !== actor.tenantId ||
lti.personId !== actor.personId ||
lti.sectionId !== resource.sectionId
) {
return deny("LTI_CONTEXT_MISMATCH");
}
if (
!lti.candidatePermissions.includes(permission) ||
!lti.deploymentPermissions.includes(permission)
) {
return deny("LTI_CAPABILITY_DENIED");
}
}
const facts = await repo.loadFacts({ actor, resource });
if (!facts.activeMembership) {
return deny("INACTIVE_MEMBERSHIP", facts.policyVersion);
}
const applicableGrants = facts.grants.filter(
grant => grant.permission === permission,
);
if (applicableGrants.length === 0) {
return deny("PERMISSION_MISSING", facts.policyVersion);
}
if (!PURPOSES[permission].includes(purpose)) {
return deny("PURPOSE_MISMATCH", facts.policyVersion);
}
if (
!relationshipAllows(permission, actor, resource, facts)
) {
return deny("RELATIONSHIP_MISSING", facts.policyVersion);
}
const permittedFields = new Set(
applicableGrants.flatMap(grant => grant.allowedFields),
);
if (
input.requestedFields.some(
field => !permittedFields.has(field),
)
) {
return deny("FIELD_SCOPE_DENIED", facts.policyVersion);
}
return {
allow: true,
reason: "ALLOWED",
permittedFields: [...new Set(input.requestedFields)],
policyVersion: facts.policyVersion,
};
}

Integration requirements
FERPA requires reasonable methods to restrict school officials to records in which they have
legitimate educational interests. The relationship and field checks help implement that
requirement.

A retention trigger should block premature destruction, not delete disclosure records
automatically.
Section 99.32 requires the applicable request/disclosure record to be maintained for as long as
the studentʼs education records are maintained. Section 99.10 also prohibits destruction of
education records while an inspection request is outstanding.
Therefore:
Education records remain → preserve required disclosure history
Outstanding inspection → preserve applicable records
Legal/institutional hold → preserve
Unknown retirement state → preserve
Approved retirement + no holds → eligible for reviewed disposition
Operational security logs can have a different approved schedule. Do not apply that schedule to
statutory disclosure records.
