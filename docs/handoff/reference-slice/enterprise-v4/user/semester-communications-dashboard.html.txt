          <div class="proto-row"><strong>Join</strong><div><code>SUBSCRIBE_CONVERSATION { conversationId, cursor }</code> attaches presence, unread state, and backlog cursor.</div></div>
          <div class="proto-row"><strong>Send</strong><div><code>SEND_MESSAGE { clientMessageId, conversationId, body, attachments[], replyToId }</code> returns ack, server message id, and moderation state.</div></div>
          <div class="proto-row"><strong>Events</strong><div><code>MESSAGE_CREATED</code>, <code>MESSAGE_READ</code>, <code>TYPING_STARTED</code>, <code>PRESENCE_UPDATED</code>, <code>CONVERSATION_MUTED</code>, <code>MESSAGE_FLAGGED</code>.</div></div>
          <div class="proto-row"><strong>Guarantees</strong><div>Use idempotent <code>clientMessageId</code>, ordered sequence per conversation, resumable delivery cursor, and server-side policy checks before fan-out.</div></div>

        <div class="eyebrow">TypeScript notification hub</div>
        <pre>type NotificationEventType =
  | "announcement.published"
  | "announcement.required_read"
  | "message.created"
  | "message.mentioned"
  | "conversation.invited"
  | "profile.visibility_changed";

interface NotificationEvent {
  id: string;
  type: NotificationEventType;
  actorUserId: string;
  audience: AudienceRule[];
  payload: Record&lt;string, unknown&gt;;
  channels: ("in_app" | "email" | "push" | "webhook")[];
  createdAt: string;
}

interface AudienceRule {
  kind: "role" | "cohort" | "org" | "program" | "user" | "segment";
  operator: "in" | "not_in" | "equals";
  values: string[];
}

interface NotificationHub {
  publish(event: NotificationEvent): Promise&lt;void&gt;;
  resolveAudience(eventId: string): Promise&lt;string[]&gt;;
  fanOut(eventId: string): Promise&lt;void&gt;;
  markRead(userId: string, deliveryId: string): Promise&lt;void&gt;;
}</pre>
