import { Router, type Request, type Response, type NextFunction } from "express";
import { createClient, type SupabaseClient } from "@supabase/supabase-js";
import { z } from "zod";

// Express reference routes for the prior sx/sx_private/sx_personal schema.
// Use bearer auth; configure sx/sx_personal API exposure deliberately. NEVER use service_role here.
const UUID = z.string().uuid();
const NoteBody = z.object({ title: z.string().trim().min(1).max(200), body: z.string().max(100000) }).strict();
const CheckoutBody = z.object({ plan: z.enum(["plus", "pro"]), cadence: z.enum(["monthly", "yearly"]), returnPath: z.enum(["/app/student", "/account/subscription"]) }).strict();
const ChangeBody = z.object({ plan: z.enum(["free", "plus", "pro"]), cadence: z.enum(["monthly", "yearly"]).optional() }).strict().superRefine((v, ctx) => { if (v.plan !== "free" && !v.cadence) ctx.addIssue({ code: "custom", path: ["cadence"], message: "Paid plans require cadence" }); });
const InstitutionBody = z.object({ institutionName: z.string().trim().min(1).max(200), contactEmail: z.string().email(), enrolledStudents: z.number().int().positive().max(10000000), requestedModules: z.array(z.enum(["learning", "advising", "registration", "student_accounts", "reporting"])).min(1).max(5) }).strict();
const Query = z.object({ limit: z.coerce.number().int().min(1).max(100).default(25), offset: z.coerce.number().int().min(0).max(10000).default(0) }).strict();
type Identity = { userId: string; client: SupabaseClient; requestId: string };
export interface Commerce {
 // Adapters MUST authorize ownership, persist idempotency, resolve immutable catalog prices,
 // transact local records, and use trusted tokenizing rails. Return destinations from approved config.
 createCheckout(input: { actorId: string; key: string; plan: "plus" | "pro"; cadence: "monthly" | "yearly"; returnPath: string }): Promise<{ checkoutId: string; checkoutUrl: string; state: "pending" }>;
 scheduleChange(input: { actorId: string; subscriptionId: string; key: string; plan: "free" | "plus" | "pro"; cadence?: "monthly" | "yearly" }): Promise<{ changeId: string; effectiveAt: string; paymentRequired: boolean }>;
 institutionRequest(input: { actorId: string; key: string; institutionName: string; contactEmail: string; enrolledStudents: number; requestedModules: string[] }): Promise<{ requestId: string; state: "submitted" }>;
 membership(actorId: string): Promise<{ personalPlan: "free" | "plus" | "pro"; subscriptions: unknown[]; sponsorships: unknown[]; entitlements: unknown[] }>;
}
export interface Config { url: string; publishableKey: string; commerce?: Commerce; requireApprovedCheckoutUrl(url: string): void }
class HttpError extends Error { constructor(public status: number, public code: string, message: string) { super(message); } }
function dbError(error: {code?: string} | null) { if (!error) return; if (error.code === "42501") throw new HttpError(403,"FORBIDDEN","Action not permitted"); if (error.code === "23505") throw new HttpError(409,"CONFLICT","Record conflict"); throw new HttpError(503,"DATABASE_UNAVAILABLE","Database request failed"); }
function asyncRoute(fn: (req: Request,res: Response) => Promise<void>) { return (req: Request,res: Response,next: NextFunction) => { void fn(req,res).catch(next); }; }
function id(req: Request,name: string) { return UUID.parse(req.params[name]); }
function key(req: Request) { return UUID.parse(req.header("Idempotency-Key")); }
function identity(res: Response): Identity { return res.locals.identity as Identity; }
function success(res: Response,data: unknown,status=200) { res.status(status).json({ok:true,data,requestId:identity(res).requestId}); }
export function makeSemesterRouter(config: Config) {
 const router = Router();
 router.use((req,res,next) => { res.locals.requestId=crypto.randomUUID(); res.setHeader("Cache-Control","no-store"); next(); });
 router.use((req,res,next) => {
  void (async()=>{
   const match=/^Bearer ([^\s]+)$/.exec(req.header("Authorization") ?? "");
   if(!match) throw new HttpError(401,"UNAUTHENTICATED","Bearer token required");
   const token=match[1];
   const client=createClient(config.url,config.publishableKey,{global:{headers:{Authorization:`Bearer ${token}`}},auth:{persistSession:false,autoRefreshToken:false}});
   const {data,error}=await client.auth.getUser(token);
   if(error || !data.user) throw new HttpError(401,"UNAUTHENTICATED","Invalid session");
   res.locals.identity={userId:data.user.id,client,requestId:res.locals.requestId};
   next();
  })().catch(next);
 });
 const commerce=()=>{if(!config.commerce) throw new HttpError(501,"NOT_IMPLEMENTED","Commerce adapter is not configured");return config.commerce;};
 router.get("/me/sections",asyncRoute(async(req,res)=>{
  const {limit,offset}=Query.parse(req.query);const ctx=identity(res);
  const {data,error}=await ctx.client.schema("sx").from("sections").select("id,tenant_id,title,term_id,course_version_id").order("id").range(offset,offset+limit-1);
  dbError(error);success(res,{items:data??[],offset,limit});
 }));
 router.get("/tenants/:tenantId/sections/:sectionId/assignments",asyncRoute(async(req,res)=>{
  const tenant=id(req,"tenantId"),section=id(req,"sectionId"),{limit,offset}=Query.parse(req.query);const ctx=identity(res);
  const {data,error}=await ctx.client.schema("sx").from("assignments").select("id,title,instructions,due_at,published").eq("tenant_id",tenant).eq("section_id",section).order("id").range(offset,offset+limit-1);
  dbError(error);success(res,{items:data??[],offset,limit});
 }));
 router.get("/me/grades",asyncRoute(async(req,res)=>{
  const ctx=identity(res),{limit,offset}=Query.parse(req.query);
  const enrollment=await ctx.client.schema("sx").from("enrollments").select("id").eq("student_id",ctx.userId);dbError(enrollment.error);
  const ids=(enrollment.data??[]).map(e=>e.id as string);
  if(!ids.length){success(res,{items:[],offset,limit});return;}
  const result=await ctx.client.schema("sx").from("grades").select("id,tenant_id,section_id,enrollment_id,score,published").in("enrollment_id",ids).eq("published",true).order("id").range(offset,offset+limit-1);
  dbError(result.error);success(res,{items:result.data??[],offset,limit});
 }));
 router.get("/me/notes",asyncRoute(async(req,res)=>{
  const ctx=identity(res),{limit,offset}=Query.parse(req.query);
  const result=await ctx.client.schema("sx_personal").from("notes").select("id,title,body,created_at").eq("owner_id",ctx.userId).order("id").range(offset,offset+limit-1);
  dbError(result.error);success(res,{items:result.data??[],offset,limit});
 }));
 router.post("/me/notes",asyncRoute(async(req,res)=>{
  const body=NoteBody.parse(req.body),ctx=identity(res);
  const result=await ctx.client.schema("sx_personal").rpc("create_note",{p_key:key(req),p_title:body.title,p_body:body.body});dbError(result.error);success(res,result.data,201);
 }));
 router.get("/tenants/:tenantId/invoices/:invoiceId",asyncRoute(async(req,res)=>{
  const result=await identity(res).client.schema("sx").from("invoices").select("id,number,currency,status,subtotal,tax,discount,total,due_at").eq("tenant_id",id(req,"tenantId")).eq("id",id(req,"invoiceId")).maybeSingle();
  dbError(result.error);if(!result.data) throw new HttpError(404,"NOT_FOUND","Invoice not found");success(res,result.data);
 }));
 router.get("/me/membership",asyncRoute(async(req,res)=>{success(res,await commerce().membership(identity(res).userId));}));
 router.post("/checkout",asyncRoute(async(req,res)=>{
  const body=CheckoutBody.parse(req.body);
  const data=await commerce().createCheckout({actorId:identity(res).userId,key:key(req),...body});
  config.requireApprovedCheckoutUrl(data.checkoutUrl);success(res,data,201);
 }));
 router.post("/subscriptions/:subscriptionId/change",asyncRoute(async(req,res)=>{
  const body=ChangeBody.parse(req.body);
  success(res,await commerce().scheduleChange({actorId:identity(res).userId,subscriptionId:id(req,"subscriptionId"),key:key(req),...body}),202);
 }));
 router.post("/institution-requests",asyncRoute(async(req,res)=>{
  const body=InstitutionBody.parse(req.body);
  success(res,await commerce().institutionRequest({actorId:identity(res).userId,key:key(req),...body}),202);
 }));
 router.use((error: unknown,_req: Request,res: Response,_next: NextFunction)=>{
  const requestId=res.locals.requestId;
  if(error instanceof z.ZodError){res.status(400).json({ok:false,error:{code:"INVALID_REQUEST",message:"Invalid input",issues:error.issues,requestId}});return;}
  if(error instanceof HttpError){res.status(error.status).json({ok:false,error:{code:error.code,message:error.message,requestId}});return;}
  res.status(500).json({ok:false,error:{code:"INTERNAL_ERROR",message:"Request could not be completed",requestId}});
 });
 return router;
}
