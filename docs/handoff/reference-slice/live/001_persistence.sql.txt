-- Standalone Supabase reference: reconcile with existing schemas BEFORE use.
-- Run as trusted migration owner. Server pg connection is privileged and MUST
-- validate authenticated actors and tenant permissions for every public operation.
BEGIN;
CREATE SCHEMA semester_live;
CREATE SCHEMA semester_live_private;
REVOKE ALL ON SCHEMA semester_live,semester_live_private FROM PUBLIC,anon,authenticated;
GRANT USAGE ON SCHEMA semester_live,semester_live_private TO authenticated;
CREATE TABLE semester_live.tenants(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),name text NOT NULL,status text NOT NULL DEFAULT 'active' CHECK(status IN ('active','suspended','closed')));
CREATE TABLE semester_live_private.memberships(tenant_id uuid NOT NULL REFERENCES semester_live.tenants(id),user_id uuid NOT NULL REFERENCES auth.users(id),active boolean NOT NULL DEFAULT true,PRIMARY KEY(tenant_id,user_id));
CREATE TABLE semester_live_private.roles(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),tenant_id uuid NOT NULL REFERENCES semester_live.tenants(id),code text NOT NULL,UNIQUE(tenant_id,id),UNIQUE(tenant_id,code));
CREATE TABLE semester_live_private.role_permissions(role_id uuid NOT NULL REFERENCES semester_live_private.roles(id),permission text NOT NULL,PRIMARY KEY(role_id,permission));
CREATE TABLE semester_live_private.assignments(tenant_id uuid NOT NULL,user_id uuid NOT NULL,role_id uuid NOT NULL,expires_at timestamptz,FOREIGN KEY(tenant_id,user_id) REFERENCES semester_live_private.memberships(tenant_id,user_id),FOREIGN KEY(tenant_id,role_id) REFERENCES semester_live_private.roles(tenant_id,id),PRIMARY KEY(tenant_id,user_id,role_id));
CREATE TABLE semester_live.documents(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),tenant_id uuid NOT NULL REFERENCES semester_live.tenants(id),title text NOT NULL,document_type text NOT NULL,created_by uuid NOT NULL REFERENCES auth.users(id),legal_hold boolean NOT NULL DEFAULT false,created_at timestamptz NOT NULL DEFAULT now(),UNIQUE(tenant_id,id));
CREATE TABLE semester_live.versions(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),tenant_id uuid NOT NULL,document_id uuid NOT NULL,version_number integer NOT NULL CHECK(version_number>0),state text NOT NULL CHECK(state IN ('draft','in_review','changes_requested','approved','awaiting_signature','effective','rejected','superseded','archived')),snapshot jsonb NOT NULL CHECK(jsonb_typeof(snapshot)='object'),storage_key text NOT NULL,content_hash text NOT NULL CHECK(content_hash ~ '^[0-9a-f]{64}$'),created_at timestamptz NOT NULL DEFAULT now(),FOREIGN KEY(tenant_id,document_id) REFERENCES semester_live.documents(tenant_id,id),CHECK(snapshot->>'state'=state),UNIQUE(tenant_id,id),UNIQUE(document_id,version_number));
CREATE UNIQUE INDEX single_effective_version ON semester_live.versions(document_id) WHERE state='effective';
CREATE TABLE semester_live.events(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),tenant_id uuid NOT NULL,version_id uuid NOT NULL,actor_id uuid NOT NULL REFERENCES auth.users(id),action text NOT NULL,from_state text,to_state text NOT NULL,reason text NOT NULL,created_at timestamptz NOT NULL DEFAULT now(),FOREIGN KEY(tenant_id,version_id) REFERENCES semester_live.versions(tenant_id,id));
CREATE TABLE semester_live_private.receipts(tenant_id uuid NOT NULL REFERENCES semester_live.tenants(id),actor_id uuid NOT NULL REFERENCES auth.users(id),request_key uuid NOT NULL,payload jsonb NOT NULL,response jsonb NOT NULL,PRIMARY KEY(tenant_id,actor_id,request_key));
CREATE TABLE semester_live_private.signatures(tenant_id uuid NOT NULL,version_id uuid NOT NULL,signer_id uuid NOT NULL REFERENCES auth.users(id),content_hash text NOT NULL,provider_reference text NOT NULL UNIQUE,verified_at timestamptz NOT NULL DEFAULT now(),FOREIGN KEY(tenant_id,version_id) REFERENCES semester_live.versions(tenant_id,id),PRIMARY KEY(tenant_id,version_id,signer_id));
CREATE TABLE semester_live_private.endpoints(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),tenant_id uuid NOT NULL REFERENCES semester_live.tenants(id),url text NOT NULL,secret_ref text NOT NULL,event_types text[] NOT NULL,enabled boolean NOT NULL DEFAULT true,UNIQUE(tenant_id,id));
CREATE TABLE semester_live_private.outbox(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),tenant_id uuid NOT NULL REFERENCES semester_live.tenants(id),event_type text NOT NULL,body text NOT NULL,created_at timestamptz NOT NULL DEFAULT now(),UNIQUE(tenant_id,id));
CREATE TABLE semester_live.deliveries(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),tenant_id uuid NOT NULL,event_id uuid NOT NULL,endpoint_id uuid NOT NULL,status text NOT NULL DEFAULT 'pending' CHECK(status IN ('pending','processing','retry_wait','delivered','dead_letter','dismissed')),attempts integer NOT NULL DEFAULT 0 CHECK(attempts>=0),replays integer NOT NULL DEFAULT 0 CHECK(replays BETWEEN 0 AND 3),due_at timestamptz NOT NULL DEFAULT now(),lease_token uuid,lease_until timestamptz,diagnostic text,updated_at timestamptz NOT NULL DEFAULT now(),FOREIGN KEY(tenant_id,event_id) REFERENCES semester_live_private.outbox(tenant_id,id),FOREIGN KEY(tenant_id,endpoint_id) REFERENCES semester_live_private.endpoints(tenant_id,id),UNIQUE(event_id,endpoint_id));
CREATE INDEX delivery_due ON semester_live.deliveries(due_at) WHERE status IN ('pending','retry_wait');
CREATE TABLE semester_live_private.delivery_attempts(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),delivery_id uuid NOT NULL REFERENCES semester_live.deliveries(id),lease_token uuid NOT NULL UNIQUE,attempt integer NOT NULL,outcome text NOT NULL DEFAULT 'started',http_status integer,diagnostic text,created_at timestamptz NOT NULL DEFAULT now(),completed_at timestamptz);
CREATE TABLE semester_live_private.dlq_decisions(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),tenant_id uuid NOT NULL,delivery_id uuid NOT NULL REFERENCES semester_live.deliveries(id),actor_id uuid NOT NULL REFERENCES auth.users(id),decision text NOT NULL CHECK(decision IN ('replay','dismiss')),reason text NOT NULL,created_at timestamptz NOT NULL DEFAULT now());
CREATE FUNCTION semester_live_private.can(t uuid,p text) RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path='' AS $$ SELECT EXISTS(SELECT 1 FROM semester_live_private.memberships m JOIN semester_live.tenants t1 ON t1.id=m.tenant_id JOIN semester_live_private.assignments a ON a.tenant_id=m.tenant_id AND a.user_id=m.user_id JOIN semester_live_private.role_permissions rp ON rp.role_id=a.role_id WHERE m.tenant_id=t AND m.user_id=auth.uid() AND m.active AND t1.status='active' AND rp.permission=p AND (a.expires_at IS NULL OR a.expires_at>now())); $$;
DO $$ DECLARE r record; BEGIN FOR r IN SELECT schemaname,tablename FROM pg_tables WHERE schemaname IN ('semester_live','semester_live_private') LOOP EXECUTE format('ALTER TABLE %I.%I ENABLE ROW LEVEL SECURITY',r.schemaname,r.tablename); END LOOP; END $$;
CREATE POLICY documents_read ON semester_live.documents FOR SELECT TO authenticated USING(semester_live_private.can(tenant_id,'document.read'));
CREATE POLICY versions_read ON semester_live.versions FOR SELECT TO authenticated USING(semester_live_private.can(tenant_id,'document.read'));
CREATE POLICY events_read ON semester_live.events FOR SELECT TO authenticated USING(semester_live_private.can(tenant_id,'document.audit'));
CREATE POLICY deliveries_read ON semester_live.deliveries FOR SELECT TO authenticated USING(semester_live_private.can(tenant_id,'webhook.read'));
REVOKE ALL ON ALL TABLES IN SCHEMA semester_live,semester_live_private FROM PUBLIC,anon,authenticated;
GRANT SELECT ON semester_live.documents,semester_live.versions,semester_live.events,semester_live.deliveries TO authenticated;
REVOKE ALL ON FUNCTION semester_live_private.can(uuid,text) FROM PUBLIC,anon;
GRANT EXECUTE ON FUNCTION semester_live_private.can(uuid,text) TO authenticated;
COMMIT;
