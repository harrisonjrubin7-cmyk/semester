import test from 'node:test';
import assert from 'node:assert/strict';
import {transition,WorkflowError} from '../transition.js';
const doc=(state='draft',overrides={})=>({state,authorId:'author',reviewerIds:['a','b'],approvalIds:[],requiresSignature:true,signerIds:[],signedIds:[],legalHold:false,...overrides});
const cmd=(action,expectedState,participants)=>({action,expectedState,reason:'Reviewed',...(participants?{participants}:{})});
const context=(userId,permission)=>({userId,permissions:[permission]});
const error=(fn,code)=>assert.throws(fn,e=>e instanceof WorkflowError&&e.code===code);
test('submit assigns independent reviewers',()=>{const n=transition(doc(),cmd('submit','draft',['a','b']),context('author','document.submit'));assert.equal(n.state,'in_review');assert.deepEqual(n.reviewerIds,['a','b']);});
test('missing permission denied',()=>error(()=>transition(doc(),cmd('submit','draft',['a']),context('author','document.read')),'FORBIDDEN'));
test('stale expected state denied',()=>error(()=>transition(doc('in_review'),cmd('approve','draft'),context('a','document.review')),'STATE_CONFLICT'));
test('reason required',()=>error(()=>transition(doc(),{...cmd('submit','draft',['a']),reason:' '},context('author','document.submit')),'INVALID_REQUEST'));
test('duplicate reviewer denied',()=>error(()=>transition(doc(),cmd('submit','draft',['a','a']),context('author','document.submit')),'INVALID_REVIEWERS'));
test('author in review list denied',()=>error(()=>transition(doc(),cmd('submit','draft',['author']),context('author','document.submit')),'INVALID_REVIEWERS'));
test('first approval does not finish quorum',()=>assert.equal(transition(doc('in_review'),cmd('approve','in_review'),context('a','document.review')).state,'in_review'));
test('last approval completes quorum',()=>assert.equal(transition(doc('in_review',{approvalIds:['a']}),cmd('approve','in_review'),context('b','document.review')).state,'approved'));
test('author self review denied',()=>error(()=>transition(doc('in_review',{reviewerIds:['author']}),cmd('approve','in_review'),context('author','document.review')),'SELF_REVIEW'));
test('unassigned reviewer denied',()=>error(()=>transition(doc('in_review'),cmd('approve','in_review'),context('outsider','document.review')),'REVIEW_ASSIGNMENT_REQUIRED'));
test('duplicate decision denied',()=>error(()=>transition(doc('in_review',{approvalIds:['a']}),cmd('approve','in_review'),context('a','document.review')),'ALREADY_DECIDED'));
test('changes requested concludes version review',()=>assert.equal(transition(doc('in_review'),cmd('request_changes','in_review'),context('a','document.review')).state,'changes_requested'));
test('rejection concludes review',()=>assert.equal(transition(doc('in_review'),cmd('reject','in_review'),context('a','document.review')).state,'rejected'));
test('signature requests need approved signing version',()=>error(()=>transition(doc('approved',{requiresSignature:false}),cmd('request_signatures','approved',['a']),context('op','document.sign_request')),'INVALID_TRANSITION'));
test('signing request assigns signers',()=>assert.deepEqual(transition(doc('approved'),cmd('request_signatures','approved',['a']),context('op','document.sign_request')).signerIds,['a']));
test('unsigned required activation denied',()=>error(()=>transition(doc('awaiting_signature',{signerIds:['a'],signedIds:[]}),cmd('activate','awaiting_signature'),context('op','document.activate')),'EVIDENCE_REQUIRED'));
test('partial signature set denied',()=>error(()=>transition(doc('awaiting_signature',{signerIds:['a','b'],signedIds:['a']}),cmd('activate','awaiting_signature'),context('op','document.activate')),'EVIDENCE_REQUIRED'));
test('all signatures allow activation',()=>assert.equal(transition(doc('awaiting_signature',{signerIds:['a','b'],signedIds:['a','b']}),cmd('activate','awaiting_signature'),context('op','document.activate')).state,'effective'));
test('no-signature approved version activates',()=>assert.equal(transition(doc('approved',{requiresSignature:false}),cmd('activate','approved'),context('op','document.activate')).state,'effective'));
test('held archive denied',()=>error(()=>transition(doc('superseded',{legalHold:true}),cmd('archive','superseded'),context('op','document.archive')),'LEGAL_HOLD'));
test('effective direct archive denied',()=>error(()=>transition(doc('effective'),cmd('archive','effective'),context('op','document.archive')),'INVALID_TRANSITION'));
test('superseded archives',()=>assert.equal(transition(doc('superseded'),cmd('archive','superseded'),context('op','document.archive')).state,'archived'));
test('unknown action denied',()=>error(()=>transition(doc(),cmd('erase','draft'),context('op','document.archive')),'INVALID_ACTION'));
test('transition does not mutate input',()=>{const d=doc('in_review');const before=structuredClone(d);transition(d,cmd('approve','in_review'),context('a','document.review'));assert.deepEqual(d,before);});
