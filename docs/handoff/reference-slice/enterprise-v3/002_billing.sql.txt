-- 002_billing.sql: run after 001_core.sql. Amounts are integer minor units.
BEGIN;
SET LOCAL lock_timeout='5s';
CREATE TABLE sem_billing.accounts(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),owner_tenant_id uuid NOT NULL REFERENCES sem_core.tenants(id),name text NOT NULL,status text NOT NULL CHECK(status IN ('active','closed')),UNIQUE(owner_tenant_id,id));
CREATE TABLE sem_billing.account_tenants(account_id uuid NOT NULL REFERENCES sem_billing.accounts(id),tenant_id uuid NOT NULL REFERENCES sem_core.tenants(id),PRIMARY KEY(account_id,tenant_id));
CREATE TABLE sem_private.account_access(account_id uuid NOT NULL,tenant_id uuid NOT NULL,membership_id uuid NOT NULL,can_read boolean NOT NULL DEFAULT true,can_manage boolean NOT NULL DEFAULT false,expires_at timestamptz,PRIMARY KEY(account_id,tenant_id,membership_id),FOREIGN KEY(account_id,tenant_id) REFERENCES sem_billing.account_tenants(account_id,tenant_id),FOREIGN KEY(tenant_id,membership_id) REFERENCES sem_private.memberships(tenant_id,id));
CREATE TABLE sem_billing.products(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),owner_tenant_id uuid NOT NULL REFERENCES sem_core.tenants(id),code text NOT NULL,name text NOT NULL,UNIQUE(owner_tenant_id,code),UNIQUE(owner_tenant_id,id));
CREATE TABLE sem_billing.plans(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),owner_tenant_id uuid NOT NULL,product_id uuid NOT NULL,code text NOT NULL,FOREIGN KEY(owner_tenant_id,product_id) REFERENCES sem_billing.products(owner_tenant_id,id),UNIQUE(owner_tenant_id,code),UNIQUE(owner_tenant_id,id));
CREATE TABLE sem_billing.price_versions(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),owner_tenant_id uuid NOT NULL,plan_id uuid NOT NULL,currency text NOT NULL CHECK(currency ~ '^[A-Z]{3}$'),unit_amount bigint NOT NULL CHECK(unit_amount>=0),interval_code text NOT NULL CHECK(interval_code IN ('month','year','one_time')),effective_from timestamptz NOT NULL,effective_until timestamptz,CHECK(effective_until IS NULL OR effective_until>effective_from),FOREIGN KEY(owner_tenant_id,plan_id) REFERENCES sem_billing.plans(owner_tenant_id,id),UNIQUE(owner_tenant_id,id));
CREATE TABLE sem_billing.contracts(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL REFERENCES sem_billing.accounts(id),agreement_version_reference text NOT NULL,status text NOT NULL CHECK(status IN ('draft','review','signed','effective','terminated')),effective_from timestamptz,effective_until timestamptz,UNIQUE(account_id,id));
CREATE TABLE sem_billing.subscriptions(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,licensed_tenant_id uuid NOT NULL,contract_id uuid,status text NOT NULL CHECK(status IN ('draft','trialing','active','past_due','suspended','canceled','expired')),current_period_start timestamptz,current_period_end timestamptz,cancel_at_period_end boolean NOT NULL DEFAULT false,FOREIGN KEY(account_id,licensed_tenant_id) REFERENCES sem_billing.account_tenants(account_id,tenant_id),FOREIGN KEY(account_id,contract_id) REFERENCES sem_billing.contracts(account_id,id),CHECK(current_period_end IS NULL OR (current_period_start IS NOT NULL AND current_period_end>current_period_start)),UNIQUE(account_id,id),UNIQUE(account_id,id,licensed_tenant_id));
CREATE TABLE sem_billing.subscription_items(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,subscription_id uuid NOT NULL,catalog_owner_tenant_id uuid NOT NULL,price_version_id uuid NOT NULL,quantity bigint NOT NULL CHECK(quantity>0),FOREIGN KEY(account_id,subscription_id) REFERENCES sem_billing.subscriptions(account_id,id),FOREIGN KEY(catalog_owner_tenant_id,price_version_id) REFERENCES sem_billing.price_versions(owner_tenant_id,id));
CREATE TABLE sem_billing.entitlements(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,subscription_id uuid NOT NULL,licensed_tenant_id uuid NOT NULL,feature_code text NOT NULL,limit_value bigint CHECK(limit_value>=0),effective_from timestamptz NOT NULL,effective_until timestamptz,status text NOT NULL CHECK(status IN ('enabled','grace','disabled')),FOREIGN KEY(account_id,subscription_id,licensed_tenant_id) REFERENCES sem_billing.subscriptions(account_id,id,licensed_tenant_id),CHECK(effective_until IS NULL OR effective_until>effective_from));
CREATE TABLE sem_billing.invoices(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL REFERENCES sem_billing.accounts(id),subscription_id uuid,number text NOT NULL,currency text NOT NULL CHECK(currency ~ '^[A-Z]{3}$'),status text NOT NULL CHECK(status IN ('draft','issued','partially_paid','paid','void','uncollectible')),subtotal bigint NOT NULL CHECK(subtotal>=0),tax bigint NOT NULL CHECK(tax>=0),discount bigint NOT NULL CHECK(discount>=0),total bigint NOT NULL CHECK(total>=0),issued_at timestamptz,due_at timestamptz,CHECK(total=subtotal+tax-discount),FOREIGN KEY(account_id,subscription_id) REFERENCES sem_billing.subscriptions(account_id,id),UNIQUE(account_id,number),UNIQUE(account_id,id),UNIQUE(account_id,id,currency));
CREATE TABLE sem_billing.invoice_lines(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,invoice_id uuid NOT NULL,description text NOT NULL,quantity bigint NOT NULL CHECK(quantity>0),unit_amount bigint NOT NULL CHECK(unit_amount>=0),line_total bigint NOT NULL CHECK(line_total=quantity*unit_amount),FOREIGN KEY(account_id,invoice_id) REFERENCES sem_billing.invoices(account_id,id));
CREATE TABLE sem_private.payment_methods(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL REFERENCES sem_billing.accounts(id),rail text NOT NULL,token_reference text NOT NULL,display_last4 text CHECK(display_last4 ~ '^[0-9]{4}$'),status text NOT NULL CHECK(status IN ('active','revoked')),UNIQUE(account_id,id));
CREATE TABLE sem_billing.payment_intents(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,invoice_id uuid NOT NULL,payment_method_id uuid,status text NOT NULL CHECK(status IN ('created','requires_action','processing','succeeded','failed','canceled')),amount bigint NOT NULL CHECK(amount>0),currency text NOT NULL,idempotency_key uuid NOT NULL,FOREIGN KEY(account_id,invoice_id,currency) REFERENCES sem_billing.invoices(account_id,id,currency),FOREIGN KEY(account_id,payment_method_id) REFERENCES sem_private.payment_methods(account_id,id),UNIQUE(account_id,id),UNIQUE(account_id,id,currency),UNIQUE(account_id,idempotency_key));
CREATE TABLE sem_billing.payment_attempts(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,intent_id uuid NOT NULL,attempt_number integer NOT NULL CHECK(attempt_number>0),status text NOT NULL CHECK(status IN ('pending','succeeded','failed')),failure_code text,created_at timestamptz NOT NULL DEFAULT now(),FOREIGN KEY(account_id,intent_id) REFERENCES sem_billing.payment_intents(account_id,id),UNIQUE(intent_id,attempt_number));
CREATE TABLE sem_billing.payments(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,intent_id uuid NOT NULL,currency text NOT NULL,amount bigint NOT NULL CHECK(amount>0),status text NOT NULL CHECK(status IN ('pending','settled','failed')),rail text NOT NULL,rail_account text NOT NULL,external_reference text NOT NULL,settled_at timestamptz,FOREIGN KEY(account_id,intent_id,currency) REFERENCES sem_billing.payment_intents(account_id,id,currency),UNIQUE(rail,rail_account,external_reference),UNIQUE(account_id,id,currency));
CREATE TABLE sem_billing.payment_allocations(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,payment_id uuid NOT NULL,invoice_id uuid NOT NULL,currency text NOT NULL,amount bigint NOT NULL CHECK(amount>0),FOREIGN KEY(account_id,payment_id,currency) REFERENCES sem_billing.payments(account_id,id,currency),FOREIGN KEY(account_id,invoice_id,currency) REFERENCES sem_billing.invoices(account_id,id,currency));
CREATE TABLE sem_billing.credits(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,invoice_id uuid NOT NULL,currency text NOT NULL,amount bigint NOT NULL CHECK(amount>0),reason text NOT NULL,status text NOT NULL CHECK(status IN ('draft','approved','applied','void')),FOREIGN KEY(account_id,invoice_id,currency) REFERENCES sem_billing.invoices(account_id,id,currency));
CREATE TABLE sem_billing.refunds(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,payment_id uuid NOT NULL,currency text NOT NULL,amount bigint NOT NULL CHECK(amount>0),status text NOT NULL CHECK(status IN ('requested','approved','processing','succeeded','failed')),reason text NOT NULL,idempotency_key uuid NOT NULL,FOREIGN KEY(account_id,payment_id,currency) REFERENCES sem_billing.payments(account_id,id,currency),UNIQUE(account_id,idempotency_key));
CREATE TABLE sem_billing.reconciliation_runs(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL REFERENCES sem_billing.accounts(id),status text NOT NULL CHECK(status IN ('pending','running','completed','failed')),started_at timestamptz NOT NULL DEFAULT now(),completed_at timestamptz,UNIQUE(account_id,id));
CREATE TABLE sem_billing.reconciliation_items(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),account_id uuid NOT NULL,run_id uuid NOT NULL,external_reference text NOT NULL,outcome text NOT NULL CHECK(outcome IN ('matched','missing_local','missing_external','mismatch')),FOREIGN KEY(account_id,run_id) REFERENCES sem_billing.reconciliation_runs(account_id,id));
CREATE INDEX invoice_account_status ON sem_billing.invoices(account_id,status,due_at);
CREATE INDEX entitlement_tenant_feature ON sem_billing.entitlements(licensed_tenant_id,feature_code,status);
COMMIT;
