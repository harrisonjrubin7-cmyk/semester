export type State = "draft" | "in_review" | "changes_requested" | "approved" | "awaiting_signature" | "effective" | "rejected" | "superseded" | "archived";
type Actor = "author" | "reviewer-a" | "reviewer-b" | "operator" | "signing-integration";
type Command = "submit" | "approve" | "request_changes" | "reject" | "request_signatures" | "activate" | "supersede" | "archive";
type Doc = { state: State; version: number; author: Actor; reviewers: Actor[]; approvals: Actor[]; signaturesRequired: boolean; signaturesVerified: boolean; legalHold: boolean };
type Hook = { id: string; attempt: number; state: "pending" | "retry_wait" | "delivered" | "dead_letter"; nextDelayMs: number; error: string | null };
type Log = { at: string; message: string };
type Model = { doc: Doc; hooks: Hook[]; log: Log[]; error: string | null };
export class DomainError extends Error { constructor(public code: string, message: string) { super(message); } }
const fail = (code: string, message: string): never => { throw new DomainError(code, message); };

export function transition(doc: Doc, command: Command, actor: Actor, expected: State): Doc {
  if (doc.state !== expected) fail("STATE_CONFLICT", "Document changed; refresh before retrying.");
  const reviewer = () => {
    if (actor === doc.author) fail("SELF_REVIEW", "Authors cannot review their own version.");
    if (!doc.reviewers.includes(actor)) fail("FORBIDDEN", "An assigned reviewer is required.");
    if (doc.approvals.includes(actor)) fail("ALREADY_DECIDED", "This reviewer already approved.");
  };
  switch (command) {
    case "submit":
      if (actor !== doc.author || doc.state !== "draft") fail("FORBIDDEN", "Only the author can submit a draft.");
      if (!doc.reviewers.length || doc.reviewers.includes(doc.author)) fail("INVALID_REVIEWERS", "Independent reviewers are required.");
      return { ...doc, state: "in_review", approvals: [] };
    case "approve": {
      if (doc.state !== "in_review") fail("INVALID_TRANSITION", "Approval requires an active review.");
      reviewer();
      const approvals = [...doc.approvals, actor];
      return { ...doc, approvals, state: doc.reviewers.every(r => approvals.includes(r)) ? "approved" : "in_review" };
    }
    case "request_changes":
    case "reject":
      if (doc.state !== "in_review") fail("INVALID_TRANSITION", "An active review is required.");
      reviewer();
      return { ...doc, state: command === "reject" ? "rejected" : "changes_requested" };
    case "request_signatures":
      if (actor !== "operator" || doc.state !== "approved" || !doc.signaturesRequired) fail("INVALID_TRANSITION", "An operator can request signatures for an approved signing-required version.");
      return { ...doc, state: "awaiting_signature" };
    case "activate":
      if (actor !== "operator") fail("FORBIDDEN", "Activation requires an operator.");
      if (!((doc.state === "approved" && !doc.signaturesRequired) || (doc.state === "awaiting_signature" && doc.signaturesVerified))) fail("EVIDENCE_REQUIRED", "Approved content and any required signature evidence are necessary.");
      return { ...doc, state: "effective" };
    case "supersede":
      // Explorer-only illustration. Real SQL supersedes atomically with replacement activation.
      if (actor !== "operator" || doc.state !== "effective") fail("INVALID_TRANSITION", "An effective version is required.");
      return { ...doc, state: "superseded" };
    case "archive":
      if (actor !== "operator" || !["rejected", "superseded"].includes(doc.state)) fail("INVALID_TRANSITION", "Only rejected or superseded versions can be archived here.");
      if (doc.legalHold) fail("LEGAL_HOLD", "An active legal hold blocks archiving.");
      return { ...doc, state: "archived" };
  }
}
const initial = (): Model => ({ doc: { state: "draft", version: 1, author: "author", reviewers: ["reviewer-a", "reviewer-b"], approvals: [], signaturesRequired: true, signaturesVerified: false, legalHold: false }, hooks: [], log: [], error: null });
type Action =
 | { type: "transition"; command: Command; actor: Actor; expected: State; at: string; eventId: string }
 | { type: "verify"; actor: Actor; at: string }
 | { type: "revise"; actor: Actor; at: string }
 | { type: "hold"; value: boolean }
 | { type: "signature_required"; value: boolean }
 | { type: "hook"; id: string; status: number; at: string; jitter: number }
 | { type: "reset" };
export function reducer(model: Model, action: Action): Model {
  if (action.type === "reset") return initial();
  try {
    if (action.type === "transition") {
      const doc = transition(model.doc, action.command, action.actor, action.expected);
      const hooks = doc.state === "effective" && model.doc.state !== "effective"
        ? [...model.hooks, { id: action.eventId, attempt: 0, state: "pending" as const, nextDelayMs: 0, error: null }]
        : model.hooks;
      return { ...model, doc, hooks, error: null, log: [...model.log, { at: action.at, message: `${action.actor}: ${action.command}, ${model.doc.state} → ${doc.state}` }] };
    }
    if (action.type === "verify") {
      if (action.actor !== "signing-integration" || model.doc.state !== "awaiting_signature") fail("FORBIDDEN", "Only the simulated signing integration can attach evidence while awaiting signatures.");
      return { ...model, doc: { ...model.doc, signaturesVerified: true }, error: null, log: [...model.log, { at: action.at, message: "SIMULATED verified signature set attached; no real provider verification." }] };
    }
    if (action.type === "revise") {
      if (action.actor !== "author" || !["changes_requested", "rejected", "superseded", "archived"].includes(model.doc.state)) fail("INVALID_TRANSITION", "Author revision requires a concluded candidate in this explorer.");
      return { ...model, doc: { ...model.doc, version: model.doc.version + 1, state: "draft", approvals: [], signaturesVerified: false }, error: null, log: [...model.log, { at: action.at, message: `New content version ${model.doc.version + 1}; prior history preserved in this log.` }] };
    }
    if (action.type === "hold") return { ...model, doc: { ...model.doc, legalHold: action.value } };
    if (action.type === "signature_required") {
      if (model.doc.state !== "draft") fail("LOCKED", "Change this scenario option only in draft.");
      return { ...model, doc: { ...model.doc, signaturesRequired: action.value }, error: null };
    }
    if (action.type === "hook") {
      const hook = model.hooks.find(h => h.id === action.id);
      if (!hook || !["pending", "retry_wait"].includes(hook.state)) fail("INVALID_DELIVERY", "Delivery is already terminal or missing.");
      const attempt = hook.attempt + 1;
      const success = action.status >= 200 && action.status < 300;
      const retryable = action.status === 0 || action.status === 408 || action.status === 429 || action.status >= 500;
      const state: Hook["state"] = success ? "delivered" : retryable && attempt < 5 ? "retry_wait" : "dead_letter";
      const delay = state === "retry_wait" ? Math.floor(Math.min(60000, 1000 * 2 ** (attempt - 1)) * action.jitter) : 0;
      const next: Hook = { ...hook, attempt, state, nextDelayMs: delay, error: success ? null : action.status === 0 ? "Network failure / timeout" : `HTTP ${action.status}` };
      return { ...model, hooks: model.hooks.map(h => h.id === hook.id ? next : h), error: null, log: [...model.log, { at: action.at, message: `Webhook ${hook.id}: attempt ${attempt}, ${state}. Business document state unchanged.` }] };
    }
    return model;
  } catch (error) {
    return { ...model, error: error instanceof DomainError ? `${error.code}: ${error.message}` : "Unexpected local simulation error." };
  }
}

export const PayloadSchema = z.discriminatedUnion("type", [
  z.object({ schemaVersion: z.literal(1), type: z.literal("document.transition"), tenantId: z.string().uuid(), versionId: z.string().uuid(), idempotencyKey: z.string().uuid(), data: z.discriminatedUnion("action", [
    z.object({ action: z.literal("submit"), expectedState: z.literal("draft"), reason: z.string().trim().min(1).max(2000), reviewerIds: z.array(z.string().uuid()).min(1).max(20) }).strict(),
    z.object({ action: z.literal("approve"), expectedState: z.literal("in_review"), reason: z.string().trim().min(1).max(2000) }).strict(),
    z.object({ action: z.literal("request_signatures"), expectedState: z.literal("approved"), reason: z.string().trim().min(1).max(2000), signerIds: z.array(z.string().uuid()).min(1).max(20) }).strict(),
    z.object({ action: z.literal("activate"), expectedState: z.enum(["approved", "awaiting_signature"]), reason: z.string().trim().min(1).max(2000) }).strict()
  ]) }).strict(),
  z.object({ schemaVersion: z.literal(1), type: z.literal("support.case.create"), tenantId: z.string().uuid(), idempotencyKey: z.string().uuid(), data: z.object({ subject: z.string().trim().min(1).max(200), description: z.string().trim().min(1).max(10000), priority: z.enum(["low", "normal", "urgent"]) }).strict() }).strict()
]);

export type Stage = { name: string; minutes: number; slots: number };
export function simulate(n: number, gap: number, stages: Stage[]) {
  if (!Number.isInteger(n) || n < 1 || n > 500 || !Number.isFinite(gap) || gap <= 0) throw new Error("Use 1–500 documents and a positive arrival interval.");
  if (!stages.length || stages.some(s => !Number.isFinite(s.minutes) || s.minutes <= 0 || !Number.isInteger(s.slots) || s.slots < 1 || s.slots > 100)) throw new Error("Service times must be positive; slots must be integers from 1–100.");
  let finish = Array.from({ length: n }, (_, i) => i * gap);
  const metrics = stages.map(stage => {
    const available = Array<number>(stage.slots).fill(0);
    const waits: number[] = [];
    finish = finish.map(arrival => {
      const earliest = Math.min(...available);
      const slot = available.indexOf(earliest);
      const start = Math.max(arrival, earliest);
      waits.push(start - arrival);
      available[slot] = start + stage.minutes;
      return available[slot];
    });
    return { ...stage, capacityPerHour: 60 * stage.slots / stage.minutes, meanWait: waits.reduce((a, b) => a + b, 0) / n, maxWait: Math.max(...waits) };
  });
  const lead = finish.map((time, i) => time - i * gap);
  const bottleneck = metrics.reduce((a, b) => a.meanWait >= b.meanWait ? a : b);
  return { metrics, meanLead: lead.reduce((a, b) => a + b, 0) / n, finalCompletion: Math.max(...finish), bottleneck: bottleneck.meanWait > 0 ? bottleneck.name : "No modeled queue" };
}

    try {
      if (payload.length > 65536) throw new Error("Payload editor limit: 64 KiB characters.");
      const parsed = PayloadSchema.safeParse(JSON.parse(payload));
