-- 210 · Company and application-operations domains still proposed in the capability registry:
-- cofin (company invoices, payments, credits, expenses), people (restricted workforce), vend, sales, mkt, cms, exec, legal,
-- product, research, it, analytics, ai, dev, search, sec, ops (services, jobs, deployments, flags, backups, usage).
-- Company data is visible only to Semester staff via capabilities on the platform tenant. Not applied or tested.
create schema if not exists cofin; create schema if not exists people; create schema if not exists vend; create schema if not exists sales;
create schema if not exists mkt; create schema if not exists cms; create schema if not exists exec; create schema if not exists legal;

-- ---------- Company finance (separate from institution student finance; shares fin.ledger via posting commands only)
create table if not exists cofin.invoices(id uuid primary key default gen_random_uuid(), account_id uuid not null references biz.accounts, subscription_id uuid references commerce.subscriptions, number text unique not null, state text not null default 'draft' check (state in ('draft','issued','paid','partially_paid','void','written_off')), issued_on date, due_on date, currency text not null default 'USD', total_cents bigint not null default 0, journal_id uuid);
create table if not exists cofin.invoice_lines(id uuid primary key default gen_random_uuid(), invoice_id uuid not null references cofin.invoices on delete cascade, description text not null, quantity numeric(12,2) not null check (quantity > 0), unit_cents bigint not null, amount_cents bigint generated always as ((quantity * unit_cents)::bigint) stored);
create table if not exists cofin.payments(id uuid primary key default gen_random_uuid(), invoice_id uuid not null references cofin.invoices, amount_cents bigint not null check (amount_cents > 0), received_on date not null, method text not null, provider_ref text unique, journal_id uuid);
create table if not exists cofin.credits(id uuid primary key default gen_random_uuid(), invoice_id uuid not null references cofin.invoices, amount_cents bigint not null check (amount_cents > 0), reason text not null, approved_by uuid not null, journal_id uuid);
create table if not exists cofin.expense_claims(id uuid primary key default gen_random_uuid(), employee_id uuid not null references auth.users default auth.uid(), amount_cents bigint not null check (amount_cents > 0), category text not null, receipt_file_id uuid references comms.files, state text not null default 'submitted' check (state in ('submitted','approved','rejected','reimbursed')), approver_id uuid references auth.users, submitted_at timestamptz not null default now(), check (approver_id is null or approver_id <> employee_id));
-- Posted history can't be edited from the client; corrections are new credit/journal rows.
create or replace function cofin.deny_paid_edit() returns trigger language plpgsql as $$ begin if old.state in ('paid','void','written_off') then raise exception 'POSTED_HISTORY_IS_IMMUTABLE'; end if; return new; end $$;
drop trigger if exists inv_immutable on cofin.invoices; create trigger inv_immutable before update on cofin.invoices for each row execute function cofin.deny_paid_edit();

-- ---------- RLS: deny by default; read through capabilities on the platform tenant.
do $$ declare s text; t text; begin foreach s in array array['cofin','people','vend','sales','mkt','cms','exec','legal','product','research','it','analytics','ai','dev','search','sec'] loop
  for t in select tablename from pg_tables where schemaname = s loop execute format('alter table %I.%I enable row level security', s, t); end loop;
  execute format('revoke insert, update, delete on all tables in schema %I from anon, authenticated', s); execute format('grant usage on schema %I to authenticated', s); execute format('grant select on all tables in schema %I to authenticated', s); end loop;
  for t in select tablename from pg_tables where schemaname = 'ops' and tablename in ('services','dependencies','objectives','telemetry','alerts','builds','deployments','feature_flags','jobs','job_attempts','migrations','repair_requests','backups','restore_exercises','tenant_lifecycle_requests','usage_records') loop execute format('alter table ops.%I enable row level security', t); end loop; end $$;
