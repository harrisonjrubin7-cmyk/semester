-- 170 · Ownership, teaching authority, learning, documents, workflow, commerce.
-- From "Unified enterprise architecture map". Proposed design: reconcile with the repo before applying.
-- Principles: three explicit record scopes (personal / institutional / company) via core.workspaces;
-- course access needs a course relationship (teaching_assignments, enrollments), not an institution-wide role;
-- working grades are separate from institution-authoritative final grades; composite tenant-aware FKs;
-- signatures and approvals reference an exact immutable document version.
create schema if not exists learn; create schema if not exists docs; create schema if not exists workflow; create schema if not exists commerce;

-- ---------- Ownership
create table if not exists core.workspaces(
  id uuid primary key default gen_random_uuid(), kind text not null check (kind in ('personal','institution','company')),
  organization_id uuid references core.institutions, owner_user_id uuid references auth.users, created_at timestamptz not null default now(),
  constraint ws_shape check ((kind = 'personal' and owner_user_id is not null and organization_id is null) or (kind <> 'personal' and organization_id is not null and owner_user_id is null)));
create unique index if not exists ws_one_personal on core.workspaces(owner_user_id) where kind = 'personal';
create unique index if not exists ws_one_org on core.workspaces(organization_id, kind) where kind <> 'personal';
alter table core.workspaces enable row level security;
create policy ws_personal on core.workspaces for select using (kind = 'personal' and owner_user_id = auth.uid());
create policy ws_org on core.workspaces for select using (kind = 'institution' and organization_id = core.my_institution());

  primary key (section_id, person_id, responsibility), foreign key (section_id, institution_id) references acad.sections(id, institution_id) on delete cascade);

create or replace function acad.can_teach(p_section uuid, p_resp text[] default array['instructor_of_record','co_instructor','ta','grader','designer']) returns boolean language sql stable security definer set search_path = '' as $$
  select exists (select 1 from acad.teaching_assignments t where t.section_id = p_section and t.person_id = auth.uid() and t.responsibility = any(p_resp) and (t.ends_on is null or t.ends_on >= current_date)) $$;
create or replace function acad.enrolled(p_section uuid) returns boolean language sql stable security definer set search_path = '' as $$
  select exists (select 1 from acad.enrollments e where e.section_id = p_section and e.person_id = auth.uid() and e.role = 'student' and e.status = 'enrolled') $$;

create or replace function api.submit_work(p_assignment uuid, p_files uuid[], p_idem text) returns text language plpgsql security definer set search_path = '' as $$
declare a work.assignments; s learn.submissions; n int; code text; begin
  select * into s from learn.submissions where student_id = auth.uid() and idempotency_key = p_idem;
  if s.id is not null then return s.receipt_code; end if;                                   -- safe retry: never a duplicate submission
  select * into a from work.assignments where id = p_assignment;
  if a.id is null or not acad.enrolled(a.section_id) then raise exception 'NOT_FOUND'; end if;
  if a.state <> 'published' then raise exception 'NOT_OPEN'; end if;
  if exists (select 1 from unnest(p_files) f where not exists (select 1 from comms.files cf where cf.id = f and cf.owner_id = auth.uid() and cf.scan_state = 'clean')) then raise exception 'FILE_NOT_ALLOWED'; end if;
  select coalesce(max(attempt), 0) + 1 into n from learn.submissions where assignment_id = p_assignment and student_id = auth.uid();
  code := 'SUB-' || upper(substr(encode(extensions.gen_random_bytes(4),'hex'),1,6));
  insert into learn.submissions(assignment_id, institution_id, student_id, attempt, state, submitted_at, receipt_code, idempotency_key)
    values (p_assignment, a.institution_id, auth.uid(), n, 'submitted', now(), code, p_idem) returning * into s;
  insert into learn.submission_files(submission_id, file_id) select s.id, unnest(p_files);
  perform audit.log('submission.created', s.id::text, null, jsonb_build_object('attempt', n, 'receipt', code));
  insert into ctl.outbox(topic, tenant_id, payload) values ('submission.created', a.institution_id, jsonb_build_object('submission', s.id));
  return code; end $$;

create or replace function api.publish_module(p_module uuid) returns void language plpgsql security definer set search_path = '' as $$
declare m learn.modules; begin
  select * into m from learn.modules where id = p_module for update;
  if m.id is null or not acad.can_teach(m.section_id, array['instructor_of_record','co_instructor','designer']) then raise exception 'FORBIDDEN'; end if;
  update learn.modules set state = 'published' where id = p_module;
  perform audit.log('module.published', p_module::text, to_jsonb(m.state), '"published"');
  insert into ctl.outbox(topic, tenant_id, payload) values ('module.published', m.institution_id, jsonb_build_object('module', p_module)); end $$;

create or replace function api.record_grade(p_assignment uuid, p_student uuid, p_score numeric, p_reason text, p_from_ai boolean default false) returns uuid language plpgsql security definer set search_path = '' as $$
declare a work.assignments; g learn.grade_entries; begin
  select * into a from work.assignments where id = p_assignment;
  if a.id is null or not acad.can_teach(a.section_id, array['instructor_of_record','co_instructor','ta','grader']) then raise exception 'FORBIDDEN'; end if;
  select * into g from learn.grade_entries where assignment_id = p_assignment and student_id = p_student for update;
  if g.id is null then
    insert into learn.grade_entries(assignment_id, student_id, score, ai_suggested) values (p_assignment, p_student, p_score, p_from_ai) returning * into g;
  else
    if g.state in ('approved','synced') then raise exception 'USE_REVISION_REQUEST'; end if;
    insert into learn.grade_revisions(grade_entry_id, prior_score, new_score, reason) values (g.id, g.score, p_score, coalesce(p_reason, 'edit'));
    update learn.grade_entries set score = p_score, ai_suggested = p_from_ai, reviewed_by = case when p_from_ai then null else auth.uid() end, updated_at = now() where id = g.id;
  end if;
  perform audit.log('grade.recorded', g.id::text, null, jsonb_build_object('score', p_score, 'ai', p_from_ai)); return g.id; end $$;

create or replace function api.submit_grades_for_approval(p_assignment uuid) returns int language plpgsql security definer set search_path = '' as $$
declare a work.assignments; n int; begin
  select * into a from work.assignments where id = p_assignment;
  if a.id is null or not acad.can_teach(a.section_id, array['instructor_of_record']) then raise exception 'FORBIDDEN'; end if;
  if exists (select 1 from learn.grade_entries where assignment_id = p_assignment and ai_suggested and reviewed_by is null) then raise exception 'UNREVIEWED_AI_SUGGESTIONS'; end if;
  update learn.grade_entries set state = 'submitted_for_approval', updated_at = now() where assignment_id = p_assignment and state in ('draft','returned');
  get diagnostics n = row_count;
  insert into ctl.outbox(topic, tenant_id, payload) values ('grades.submitted', a.institution_id, jsonb_build_object('assignment', p_assignment, 'count', n));
  perform audit.log('grades.submitted', p_assignment::text, null, jsonb_build_object('count', n)); return n; end $$;
grant execute on function api.submit_work(uuid,uuid[],text), api.publish_module(uuid), api.record_grade(uuid,uuid,numeric,text,boolean), api.submit_grades_for_approval(uuid) to authenticated;

create or replace function docs.can_read(p_doc uuid) returns boolean language sql stable security definer set search_path = '' as $$
  select exists (select 1 from docs.documents d join core.workspaces w on w.id = d.workspace_id where d.id = p_doc and (
    d.classification = 'public' or (w.kind = 'personal' and w.owner_user_id = auth.uid()) or
    (w.kind = 'company' and ctl.can(case when d.classification = 'restricted' then 'docs:restricted' else 'docs:read' end, w.organization_id)) or
    (w.kind = 'institution' and w.organization_id = core.my_institution() and ctl.can('docs:read', w.organization_id)))) $$;
create policy doc_read on docs.documents for select using (docs.can_read(id));
create policy dv_read on docs.document_versions for select using (docs.can_read(document_id));
create policy ack_self on docs.acknowledgments for select using (person_id = auth.uid());

-- Versions are immutable; superseding creates a new version.
create or replace function docs.deny_version_mutation() returns trigger language plpgsql as $$ begin raise exception 'DOCUMENT_VERSIONS_ARE_IMMUTABLE'; end $$;
drop trigger if exists dv_immutable on docs.document_versions;
create trigger dv_immutable before update or delete on docs.document_versions for each row execute function docs.deny_version_mutation();
