create schema if not exists housing;
create table if not exists housing.buildings(id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, campus_id uuid references iam.campuses, name text not null, accessible boolean not null default true);
create table if not exists housing.rooms(id uuid primary key default gen_random_uuid(), building_id uuid not null references housing.buildings on delete cascade, code text not null, capacity int not null check (capacity between 1 and 8), unique (building_id, code));
create table if not exists housing.applications(id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, student_id uuid not null references core.profiles default auth.uid(), term_id uuid not null references acad.terms, preferences jsonb not null default '{}'::jsonb check (pg_column_size(preferences) < 4096), state text not null default 'submitted' check (state in ('submitted','assigned','waitlisted','withdrawn')), submitted_at timestamptz not null default now(), unique (student_id, term_id));
create table if not exists housing.assignments(id uuid primary key default gen_random_uuid(), application_id uuid not null unique references housing.applications on delete cascade, room_id uuid not null references housing.rooms, assigned_by uuid not null default auth.uid(), move_in date, source text not null default 'housing_system');
create table if not exists housing.maintenance_requests(id uuid primary key default gen_random_uuid(), room_id uuid not null references housing.rooms, requester_id uuid not null references core.profiles default auth.uid(), summary text not null check (length(summary) <= 500), urgency text not null default 'normal' check (urgency in ('normal','urgent','emergency')), state text not null default 'open' check (state in ('open','scheduled','done')), created_at timestamptz not null default now());

create or replace function api.apply_housing(p_term uuid, p_prefs jsonb) returns uuid language plpgsql security definer set search_path = '' as $$
declare a uuid; begin
  insert into housing.applications(institution_id, term_id, preferences) values (core.my_institution(), p_term, coalesce(p_prefs, '{}'))
    on conflict (student_id, term_id) do update set preferences = excluded.preferences returning id into a;
  perform audit.log('housing.applied', a::text, null, null); return a; end $$;
create or replace function api.request_maintenance(p_room uuid, p_summary text, p_urgency text) returns uuid language plpgsql security definer set search_path = '' as $$
declare m uuid; begin
  if not exists (select 1 from housing.assignments s join housing.applications a on a.id = s.application_id where s.room_id = p_room and a.student_id = auth.uid()) then raise exception 'FORBIDDEN'; end if;
  insert into housing.maintenance_requests(room_id, summary, urgency) values (p_room, p_summary, p_urgency) returning id into m;
  insert into ctl.outbox(topic, tenant_id, payload) values ('maintenance.requested', core.my_institution(), jsonb_build_object('request', m, 'urgency', p_urgency)); return m; end $$;
