-- 230 · Commands for the capabilities that still had none. Each: authorize (ctl.can / ownership) → validate → write → audit.log → ctl.outbox.
-- Personal writes (tasks, calendar, preferences, study artifacts, career) stay direct table writes under RLS from 140/200 and need no function.
-- Not applied or tested. P = platform tenant.
create or replace function api._emit(p_topic text, p_tenant uuid, p_target text, p_payload jsonb) returns void language sql security definer set search_path = '' as $$
  select audit.log(p_topic, p_target, null, p_payload); insert into ctl.outbox(topic, tenant_id, payload) values (p_topic, p_tenant, p_payload) $$;
revoke all on function api._emit(text,uuid,text,jsonb) from public, anon, authenticated;
create or replace function api._need(p_cap text, p_tenant uuid) returns void language plpgsql stable security definer set search_path = '' as $$
  begin if not ctl.can(p_cap, coalesce(p_tenant, '00000000-0000-0000-0000-000000000000')) then raise exception 'FORBIDDEN'; end if; end $$;
revoke all on function api._need(text,uuid) from public, anon, authenticated;

-- ---------- Access
create or replace function api.grant_capability(p_person uuid, p_cap text, p_tenant uuid, p_reason text, p_until timestamptz, p_resource_type text default null, p_resource_id text default null) returns uuid language plpgsql security definer set search_path = '' as $$
declare g uuid; hi boolean; begin perform api._need('grant.create', p_tenant);
  if p_person = auth.uid() then raise exception 'SELF_GRANT_FORBIDDEN'; end if;
  select high_impact into hi from ctl.capabilities where key = p_cap; if hi is null then raise exception 'NOT_FOUND'; end if;
  if hi and (p_until is null or p_until > now() + interval '90 days') then raise exception 'INVALID_REQUEST' using detail = 'high-impact grants expire within 90 days'; end if;
  insert into ctl.grants(person_id, capability, tenant_id, resource_type, resource_id, granted_by, reason, expires_at) values (p_person, p_cap, p_tenant, p_resource_type, p_resource_id, auth.uid(), p_reason, p_until) returning id into g;
  perform api._emit('grant.created', p_tenant, g::text, jsonb_build_object('person', p_person, 'cap', p_cap, 'until', p_until)); return g; end $$;
create or replace function api.revoke_grant(p_grant uuid, p_reason text) returns void language plpgsql security definer set search_path = '' as $$
declare g ctl.grants; begin select * into g from ctl.grants where id = p_grant for update; if g.id is null then raise exception 'NOT_FOUND'; end if;
  perform api._need('grant.revoke', g.tenant_id); update ctl.grants set revoked_at = now() where id = p_grant and revoked_at is null;
  perform api._emit('grant.revoked', g.tenant_id, p_grant::text, jsonb_build_object('reason', p_reason)); end $$;
create or replace function api.run_access_review(p_review uuid, p_decision text, p_reason text) returns void language plpgsql security definer set search_path = '' as $$
declare r iam.access_reviews; begin select * into r from iam.access_reviews where id = p_review for update;
  if r.id is null or r.reviewer_id <> auth.uid() then raise exception 'FORBIDDEN'; end if; if p_decision not in ('keep','revoke','reduce') then raise exception 'INVALID_REQUEST'; end if;
  update iam.access_reviews set decision = p_decision, reason = p_reason, decided_at = now() where id = p_review;
  if p_decision = 'revoke' and r.grant_id is not null then update ctl.grants set revoked_at = now() where id = r.grant_id and revoked_at is null; end if;
  perform api._emit('access_review.decided', r.institution_id, p_review::text, jsonb_build_object('decision', p_decision)); end $$;

create or replace function api.replay_job(p_job uuid, p_reason text) returns void language plpgsql security definer set search_path = '' as $$
begin perform api._need('job.replay', null); update ops.jobs set state = 'queued', next_run_at = now() where id = p_job and state in ('failed','dead'); if not found then raise exception 'INVALID_TRANSITION'; end if;
  perform audit.log('job.replayed', p_job::text, null, jsonb_build_object('reason', p_reason)); end $$;
create or replace function api.request_repair(p_desc text, p_sql_ref text) returns uuid language plpgsql security definer set search_path = '' as $$
declare r uuid; ar uuid; begin perform api._need('repair.request', null);
  insert into ctl.action_requests(capability, tenant_id, requested_by, reason, payload, idempotency_key) values ('repair.execute', '00000000-0000-0000-0000-000000000000', auth.uid(), p_desc, jsonb_build_object('sql_ref', p_sql_ref), 'repair:' || md5(p_sql_ref)) returning id into ar;
  insert into ops.repair_requests(description, sql_ref, action_request_id) values (p_desc, p_sql_ref, ar) returning id into r; return r; end $$;
create or replace function api.record_restore_test(p_backup uuid, p_rto int, p_rpo int, p_passed boolean, p_report text) returns uuid language plpgsql security definer set search_path = '' as $$
declare r uuid; begin perform api._need('console:operate', null); insert into ops.restore_exercises(backup_id, performed_on, rto_minutes, rpo_minutes, passed, report_ref) values (p_backup, current_date, p_rto, p_rpo, p_passed, p_report) returning id into r;
  update ops.backups set verified = p_passed where id = p_backup; return r; end $$;
create or replace function api.open_finding(p_source text, p_sev text, p_title text, p_due date) returns uuid language plpgsql security definer set search_path = '' as $$
declare f uuid; begin perform api._need('security:read', null); insert into sec.findings(source, severity, title, owner_id, due_on) values (p_source, p_sev, p_title, auth.uid(), p_due) returning id into f; return f; end $$;
create or replace function api.fulfil_privacy_request(p_request uuid, p_evidence text) returns void language plpgsql security definer set search_path = '' as $$
declare r opsx.privacy_requests; begin select * into r from opsx.privacy_requests where id = p_request for update; perform api._need('data_request:handle', r.institution_id);
  if r.state not in ('identity_verified','in_progress','with_institution') then raise exception 'INVALID_TRANSITION' using detail = 'verify identity first'; end if;
  update opsx.privacy_requests set state = 'fulfilled', evidence_ref = p_evidence, decided_by = auth.uid(), closed_at = now() where id = p_request; perform api._emit('privacy.fulfilled', r.institution_id, p_request::text, '{}'); end $$;
create or replace function api.verify_privacy_identity(p_request uuid, p_method text) returns void language plpgsql security definer set search_path = '' as $$
declare r opsx.privacy_requests; begin select * into r from opsx.privacy_requests where id = p_request for update; perform api._need('data_request:handle', r.institution_id);
  update opsx.privacy_requests set state = 'identity_verified', legal_basis = coalesce(legal_basis, '') || ' identity:' || p_method where id = p_request and state = 'received'; if not found then raise exception 'INVALID_TRANSITION'; end if; end $$;
create or replace function api.decide_report(p_report uuid, p_action text) returns void language plpgsql security definer set search_path = '' as $$
begin perform api._need('report.decide', core.my_institution()); if p_action not in ('dismiss','remove_content','warn','suspend') then raise exception 'INVALID_REQUEST'; end if;
  perform api._emit('report.decided', core.my_institution(), p_report::text, jsonb_build_object('action', p_action)); end $$;
create or replace function api.rotate_key(p_app uuid, p_prefix text, p_hash text) returns uuid language plpgsql security definer set search_path = '' as $$
declare k uuid; begin if not exists (select 1 from biz.partner_apps a join biz.accounts ac on ac.id = a.account_id where a.id = p_app and a.state = 'approved') then raise exception 'NOT_FOUND'; end if;
  perform api._need('app.review', null);
  update dev.credentials set revoked_at = now() + interval '24 hours' where app_id = p_app and revoked_at is null;   -- 24 h overlap for rollout
  insert into dev.credentials(app_id, key_prefix, key_hash, scopes) select p_app, p_prefix, p_hash, scopes from biz.partner_apps where id = p_app returning id into k;
  perform api._emit('key.rotated', null, p_app::text, '{}'); return k; end $$;
