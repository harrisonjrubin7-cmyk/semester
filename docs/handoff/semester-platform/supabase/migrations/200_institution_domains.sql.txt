-- ---------- iam: global identity, scoped membership (a person can belong to several institutions; permissions never merge)
create table if not exists iam.organization_units(id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, parent_id uuid references iam.organization_units, kind text not null check (kind in ('college','department','office','program','campus_unit')), name text not null, unique (id, institution_id));
create table if not exists iam.campuses(id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, name text not null, timezone text not null default 'America/Chicago');
create table if not exists iam.memberships(
  id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, person_id uuid not null references auth.users,
  affiliation text not null check (affiliation in ('student','faculty','staff','advisor','alumni','guest')), status text not null default 'active' check (status in ('invited','active','suspended','ended')),
  verified_by text, effective_from date not null default current_date, effective_until date, unique (institution_id, person_id, affiliation));
create table if not exists iam.invitations(
  id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, email text not null, affiliation text not null,
  token_hash text not null unique, invited_by uuid not null default auth.uid(), expires_at timestamptz not null default now() + interval '14 days',
  state text not null default 'pending' check (state in ('pending','accepted','revoked','expired')), accepted_by uuid references auth.users, created_at timestamptz not null default now(),
  check (expires_at <= created_at + interval '30 days'));
create unique index if not exists inv_one_pending on iam.invitations(institution_id, lower(email), affiliation) where state = 'pending';
create table if not exists iam.access_reviews(
  id uuid primary key default gen_random_uuid(), institution_id uuid references core.institutions, scope text not null, reviewer_id uuid not null references auth.users,
  grant_id uuid references ctl.grants, decision text check (decision in ('keep','revoke','reduce')), reason text, due_on date not null, decided_at timestamptz);

create policy cam_read on iam.campuses for select using (institution_id = core.my_institution());

create policy mem_admin on iam.memberships for select using (ctl.can('membership.invite', institution_id));

create or replace function api.issue_invitation(p_institution uuid, p_email text, p_affiliation text, p_token_hash text) returns uuid language plpgsql security definer set search_path = '' as $$
declare i uuid; begin
  if not ctl.can('membership.invite', p_institution) then raise exception 'FORBIDDEN'; end if;
  if p_email !~* '^[^@\s]+@[^@\s]+\.[^@\s]+$' then raise exception 'INVALID_REQUEST' using detail = 'email'; end if;
  select id into i from iam.invitations where institution_id = p_institution and lower(email) = lower(p_email) and affiliation = p_affiliation and state = 'pending';
  if i is not null then return i; end if;                                                                       -- duplicate handling
  insert into iam.invitations(institution_id, email, affiliation, token_hash) values (p_institution, lower(p_email), p_affiliation, p_token_hash) returning id into i;
  perform audit.log('invitation.issued', i::text, null, jsonb_build_object('affiliation', p_affiliation));
  insert into ctl.outbox(topic, tenant_id, payload) values ('invitation.issued', p_institution, jsonb_build_object('invitation', i)); return i; end $$;
create or replace function api.accept_invitation(p_token_hash text) returns uuid language plpgsql security definer set search_path = '' as $$
declare v iam.invitations; m uuid; begin
  select * into v from iam.invitations where token_hash = p_token_hash for update;
  if v.id is null or v.state <> 'pending' then raise exception 'NOT_FOUND'; end if;
  if v.expires_at < now() then update iam.invitations set state = 'expired' where id = v.id; raise exception 'EXPIRED'; end if;
  if lower((select email from auth.users where id = auth.uid())) <> v.email then raise exception 'FORBIDDEN' using detail = 'signed-in email does not match invitation'; end if;
  insert into iam.memberships(institution_id, person_id, affiliation, verified_by) values (v.institution_id, auth.uid(), v.affiliation, 'invitation')
    on conflict (institution_id, person_id, affiliation) do update set status = 'active' returning id into m;
  update iam.invitations set state = 'accepted', accepted_by = auth.uid() where id = v.id;
  perform audit.log('invitation.accepted', v.id::text, null, null); return m; end $$;
create or replace function api.revoke_invitation(p_invitation uuid) returns void language plpgsql security definer set search_path = '' as $$
declare v iam.invitations; begin select * into v from iam.invitations where id = p_invitation for update;
  if v.id is null or not ctl.can('membership.invite', v.institution_id) then raise exception 'FORBIDDEN'; end if;
  update iam.invitations set state = 'revoked' where id = p_invitation and state = 'pending'; perform audit.log('invitation.revoked', p_invitation::text, null, null); end $$;
grant execute on function api.issue_invitation(uuid,text,text,text), api.accept_invitation(text), api.revoke_invitation(uuid) to authenticated;

-- ---------- svc: services and cases. Restricted cases are a separate table with their own audience, never inherited.
create table if not exists svc.service_definitions(id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, office text not null, name text not null, description text, sla interval, restricted boolean not null default false, active boolean not null default true);
create table if not exists svc.service_requests(id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, service_id uuid not null references svc.service_definitions, requester_id uuid not null references core.profiles default auth.uid(), state text not null default 'submitted' check (state in ('submitted','assigned','in_progress','waiting_on_requester','resolved','closed')), assignee_id uuid references core.profiles, summary text not null check (length(summary) <= 500), sla_due timestamptz, created_at timestamptz not null default now(), resolved_at timestamptz);
create table if not exists svc.cases(id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, student_id uuid not null references core.profiles, owner_id uuid not null references core.profiles, kind text not null check (kind in ('success','outreach','referral')), state text not null default 'open' check (state in ('open','monitoring','closed')), opened_at timestamptz not null default now());
create table if not exists svc.restricted_cases(id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, student_id uuid not null references core.profiles, office text not null check (office in ('counseling','disability_services','title_ix','conduct','health')), state text not null default 'intake', created_at timestamptz not null default now());
create table if not exists svc.restricted_case_staff(case_id uuid not null references svc.restricted_cases on delete cascade, person_id uuid not null references core.profiles, purpose text not null, until timestamptz, primary key (case_id, person_id));
create table if not exists svc.accommodation_plans(id uuid primary key default gen_random_uuid(), institution_id uuid not null references core.institutions, student_id uuid not null references core.profiles, case_id uuid references svc.restricted_cases, instructions_for_faculty text not null check (length(instructions_for_faculty) <= 1000), extra_time_pct int check (extra_time_pct between 0 and 200), effective_from date not null, effective_until date, approved_by uuid not null);
do $$ declare t text; begin for t in select tablename from pg_tables where schemaname = 'svc' loop execute format('alter table svc.%I enable row level security', t); end loop; end $$;
create policy sd_read on svc.service_definitions for select using (institution_id = core.my_institution() and active);
create policy sr_self on svc.service_requests for select using (requester_id = auth.uid());
create policy sr_office on svc.service_requests for select using (exists (select 1 from svc.service_definitions d where d.id = service_id and ctl.can('request.assign', institution_id, null, 'office', d.office)));
create policy case_owner on svc.cases for select using (owner_id = auth.uid() or acad.advises(student_id));
create policy rc_self on svc.restricted_cases for select using (student_id = auth.uid());
create policy rc_staff on svc.restricted_cases for select using (exists (select 1 from svc.restricted_case_staff s where s.case_id = id and s.person_id = auth.uid() and (s.until is null or s.until > now())));
create policy rcs_staff on svc.restricted_case_staff for select using (person_id = auth.uid());
-- Faculty see only the instructions for students enrolled in a section they teach, never the case.
create policy ap_self on svc.accommodation_plans for select using (student_id = auth.uid());
create policy ap_faculty on svc.accommodation_plans for select using (exists (select 1 from acad.enrollments e join acad.teaching_assignments t on t.section_id = e.section_id where e.person_id = svc.accommodation_plans.student_id and e.status = 'enrolled' and t.person_id = auth.uid()) and (effective_until is null or effective_until >= current_date));

create policy aw_read on instfin.awards for select using (exists (select 1 from instfin.student_accounts a where a.id = account_id and (a.student_id = auth.uid() or ctl.can('award.review', a.institution_id))));
