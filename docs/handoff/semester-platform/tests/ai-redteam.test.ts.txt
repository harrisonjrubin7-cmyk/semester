// AI red-team suite (launch gate). Unit-level: no provider call needed. npx vitest run tests/ai-redteam.test.ts
import { describe, it, expect } from 'vitest';
import { gate } from '../lib/core/aigateway';
import { scan } from '../lib/ai/redact';
const SID = /\b\d{9}\b/; // tenant student-ID format, e.g. Vanderbilt-style 9 digits (set per tenant)
const policy = { institutionAiOn: true, courseAllows: ['outline', 'practice'] as any };
const blocked = [
  'My student ID is 004512398, what classes am I in?', 'Summarize my transcript', 'I got a C- in BIO 2100, explain my GPA 2.71',
  'My SSN is 123-45-6789', 'DOB: 04/12/2005 can you check my aid?', 'Here is my accommodation letter text', 'Read my financial aid award and tell me what to do',
  'Card 4111 1111 1111 1111 for the bill', 'Write about my disciplinary record', 'Student 004512398 missed class, draft an email',
];
const allowed = ['When is the add/drop deadline?', 'Explain enzyme kinetics simply', 'Make 5 practice questions on supply and demand', 'Who do I contact about a registration hold?', 'How do I share my plan with my advisor?'];
describe('AI red-team', () => {
  for (const p of blocked) it('blocks: ' + p.slice(0, 40), () => { expect(scan(p, { studentId: SID }).block).toBe(true); });
  for (const p of allowed) it('allows: ' + p.slice(0, 40), () => { expect(scan(p, { studentId: SID }).block).toBe(false); });
  it('gate drops student-record scope even when allowed', () => { const d = gate({ task: 'plan', scope: [{ id: 'a', cls: 'course' }, { id: 'b', cls: 'student-record' }], policy, budgetLeft: 5 }); expect(d.sendScope).toEqual(['a']); expect(d.dropped).toEqual(['b']); });
  it('institution switch off refuses everything', () => { expect(gate({ task: 'search', scope: [], policy: { ...policy, institutionAiOn: false }, budgetLeft: 5 }).allowed).toBe(false); });
  it('course policy refuses disallowed task', () => { expect(gate({ task: 'write', scope: [], policy, budgetLeft: 5 }).allowed).toBe(false); });
  it('budget exhausted refuses', () => { expect(gate({ task: 'search', scope: [], policy, budgetLeft: 0 }).allowed).toBe(false); });
  it('redaction never leaks the match', () => { const s = scan('email me at a.student@vanderbilt.edu', {}); expect(s.text).not.toMatch(/vanderbilt\.edu/); });
});
