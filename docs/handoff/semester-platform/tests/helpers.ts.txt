// tests/helpers.ts — signed-in and anonymous clients against local Supabase.
import { createClient, type SupabaseClient } from '@supabase/supabase-js';
import 'dotenv/config';
const url = process.env.SUPABASE_URL!, key = process.env.SUPABASE_ANON_KEY!;
const cache = new Map<string, SupabaseClient>();
export async function asUser(email: string): Promise<SupabaseClient> {
  if (cache.has(email)) return cache.get(email)!;
  const c = createClient(url, key, { auth: { persistSession: false } });
  const { error } = await c.auth.signInWithPassword({ email, password: process.env.TEST_PASSWORD! });
  if (error) throw new Error('seed user missing: ' + email + ' — run npx supabase db reset');
  cache.set(email, c); return c;
}
export function anon(): SupabaseClient { return createClient(url, key, { auth: { persistSession: false } }); }
