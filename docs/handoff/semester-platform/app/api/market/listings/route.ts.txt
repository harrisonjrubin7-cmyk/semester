import { NextResponse } from 'next/server';
import { createClient } from '@/lib/supabase/server';
import { Listing, integrityPrecheck } from '@/lib/validate';
export async function POST(req: Request) {
  const parsed = Listing.safeParse(await req.json());
  if (!parsed.success) return NextResponse.json({ error: 'Check the listing details', issues: parsed.error.flatten() }, { status: 400 });
  const sb = await createClient();
  const { data: { user } } = await sb.auth.getUser();
  if (!user) return NextResponse.json({ error: 'Sign in to sell' }, { status: 401 });
  const pre = integrityPrecheck(parsed.data.title);
  const { data, error } = await sb.schema('market').from('listings').insert({ title: parsed.data.title, category: parsed.data.category, price_cents: parsed.data.priceCents }).select().single();
  if (error) return NextResponse.json({ error: 'Your campus may not have the marketplace turned on.' }, { status: 403 });
  await sb.rpc('log', { a: 'listing.create', t: data.id, b: null, af: { ...data, precheck: pre } });
  return NextResponse.json({ listing: data, message: pre.hold ? 'Submitted. It will be reviewed before going live.' : 'Submitted. Live after the automatic check.' }, { status: 201 });
}
