async transition(ctx: TransitionContext, action: Action): Promise<State> {
const machine = await this.repo.getMachine();
const wf = await this.repo.getWorkflow(ctx.workflowInstanceId);
const stateDef = machine.states[wf.currentState];
if (!stateDef) throw new TransitionError("Unknown state", "STATE_UNKNOWN");
const candidates = stateDef.transitions.filter(
t => t.action === action && t.actors.includes(ctx.actorRole)
);
if (!candidates.length) {
throw new TransitionError("Action not permitted", "ACTION_NOT_ALLOWED");
}
let selected: TransitionDef | undefined;
for (const t of candidates) {
const guardsOk = await this.evaluateGuards(ctx, t.guards ?? []);
if (!guardsOk) continue;
const schema = await this.validator.validateTransitionPayload({
action,
fromState: wf.currentState,
toState: t.to,
context: ctx
});
if (!schema.ok) {
throw new TransitionError(
`Schema validation failed: ${schema.issues.map(i => i.code).join(",")}`,
"SCHEMA_INVALID"
);
}
selected = t;
break;
}
if (!selected) {
throw new TransitionError("No valid transition found", "TRANSITION_INVALID");
}
const now = new Date().toISOString();
const payload = {
workflowInstanceId: ctx.workflowInstanceId,
entityId: ctx.entityId,
fromState: wf.currentState,
toState: selected.to,
action,
actorUserId: ctx.actorUserId,
actorRole: ctx.actorRole,
scopeType: ctx.scopeType,
scopeId: ctx.scopeId,
requestId: ctx.requestId,
comment: ctx.comment ?? null,
metadata: ctx.metadata ?? {},
createdAt: now
};
await this.repo.inTransaction(async () => {
const previousHash = await this.repo.getLatestLedgerHash(ctx.workflowInstanceId
const payloadJson = stableStringify(payload);
const entryHash = hashTransition(previousHash, payloadJson);
const signature = signHash(entryHash, this.signingSecret);
await this.repo.updateWorkflowState(
ctx.workflowInstanceId,
wf.version,
selected!.to
);
await this.repo.insertLedgerRow({
id: crypto.randomUUID(),
workflowInstanceId: ctx.workflowInstanceId,
entityId: ctx.entityId,
fromState: wf.currentState,
toState: selected!.to,
action,
actorUserId: ctx.actorUserId,
actorRole: ctx.actorRole,
scopeType: ctx.scopeType,
scopeId: ctx.scopeId,
requestId: ctx.requestId,
payloadJson,
payloadSchemaVersion: 1,
previousHash,
entryHash,
signature,
createdAt: now
});
await this.repo.insertAuditEvent({
actorUserId: ctx.actorUserId,
actorRole: ctx.actorRole,
module: "registrar_workflow",
action: `transition.${action}`,
targetType: "workflow_instance",
targetId: ctx.workflowInstanceId,
scopeType: ctx.scopeType,
scopeId: ctx.scopeId,
requestId: ctx.requestId,
ipAddress: ctx.ipAddress,
userAgent: ctx.userAgent,
metadata: {
fromState: wf.currentState,
toState: selected!.to,
entityId: ctx.entityId,
entryHash
}
});
const event: DomainEvent = {
id: crypto.randomUUID(),
type: mapActionToEventType(action, selected!.to),
aggregateType: "workflow_instance",
aggregateId: ctx.workflowInstanceId,
payload: {
entityId: ctx.entityId,
fromState: wf.currentState,
toState: selected!.to,
actorUserId: ctx.actorUserId,
actorRole: ctx.actorRole,
requestId: ctx.requestId
},
occurredAt: now
};
await this.repo.insertOutboxEvent(event);
});
if (selected.hooks?.includes("queueValidation")) {
await this.hooks.queueValidation(ctx);
}
if (selected.hooks?.includes("createWorkflowTask")) {
await this.hooks.createWorkflowTask(ctx, selected.to);
}
return selected.to;
}
private async evaluateGuards(ctx: TransitionContext, guards: string[]) {
for (const name of guards) {
const fn = this.guards[name as keyof GuardService];
if (!fn) throw new TransitionError(`Unknown guard: ${name}`, "GUARD_UNKNOWN");
const ok = await fn.call(this.guards, ctx);
if (!ok) return false;
}
return true;
}
}
export function hashTransition(previousHash: string | null, payloadJson: string) {
return crypto
.createHash("sha256")
.update(`${previousHash ?? "GENESIS"}|${payloadJson}`)
.digest("hex");
}
export function signHash(entryHash: string, secret: string) {
return crypto.createHmac("sha256", secret).update(entryHash).digest("hex");
}
export function verifyLedgerRow(input: {
previousHash: string | null;
payloadJson: string;
entryHash: string;
signature: string;
secret: string;
}) {
const computedHash = hashTransition(input.previousHash, input.payloadJson);
const computedSig = signHash(computedHash, input.secret);
return computedHash === input.entryHash && computedSig === input.signature;
}
export function stableStringify(value: unknown): string {
return JSON.stringify(sortKeys(value));
}
function sortKeys(value: unknown): unknown {
if (Array.isArray(value)) return value.map(sortKeys);
if (value && typeof value === "object") {
return Object.keys(value as Record<string, unknown>)
.sort()
.reduce<Record<string, unknown>>((acc, key) => {
acc[key] = sortKeys((value as Record<string, unknown>)[key]);
return acc;
}, {});
}
return value;
}
function mapActionToEventType(action: Action, toState: State): string {
if (action === "publish") return "term.schedule_published";
if (toState === "academic_affairs_review") return "section.exception_routed";
if (toState === "validation_failed") return "section.validation_failed";
return `workflow.${action}`;
}

export interface WebhookDelivery {
id: string;
callbackUrl: string;
payload: Record<string, unknown>;
signingSecret: string;
attemptCount: number;
eventId: string;
eventType: string;
}
export class WebhookDeliveryService {
async deliver(job: WebhookDelivery) {
const body = JSON.stringify(job.payload);
const timestamp = new Date().toISOString();
const signature = crypto
.createHmac("sha256", job.signingSecret)
.update(`${timestamp}.${body}`)
.digest("hex");
const response = await fetch(job.callbackUrl, {
method: "POST",
headers: {
"content-type": "application/json",
"x-semester-event-id": job.eventId,
"x-semester-event-type": job.eventType,
"x-semester-timestamp": timestamp,
"x-semester-signature": signature,
"idempotency-key": `${job.eventId}:${job.attemptCount + 1}`
},
body
});
if (response.status >= 200 && response.status < 300) {
return { status: "sent", httpStatus: response.status };
}
if (response.status >= 400 && response.status < 500 && response.status !== 429) {
return { status: "quarantined", httpStatus: response.status };
}
return {
status: "retry",
httpStatus: response.status,
nextAttemptAt: computeNextAttemptAt(job.attemptCount + 1)
};
}
}
export function computeNextAttemptAt(attempt: number) {
const delays = [0, 30, 120, 600, 3600, 21600, 86400];
const seconds = delays[Math.min(attempt, delays.length - 1)];
return new Date(Date.now() + seconds * 1000).toISOString();
}

Cryptographic verification should happen at three points:
On write, generate entry_hash and HMAC signature.
On read/export, recompute and verify each ledger row.
On periodic audit job, verify full chain from genesis to tip and alert on any gap or
mismatch.
That gives you a tamper-evident workflow ledger rather than a plain audit table, which is
materially stronger for registrar and FERPA-sensitive operations.
Validation hooks should be split into:
Schema validation: payload shape, required keys, enum values.
Guard validation: role, state, exception eligibility, comment requirements.
Domain validation: meeting patterns, overlap rules, room capacity, publish readiness.
Compliance validation: legitimate-interest presence for sensitive exports or overrides.
This structure matches both approval workflow concepts and FERPA-oriented access-control
expectations from the sources.
