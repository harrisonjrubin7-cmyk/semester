    <div className="tiles"><N.MetricTile label="Read keys" value={String(capRows.filter((c) => c.kind === 'read').length)} /><N.MetricTile label="Write keys" value={String(capRows.filter((c) => c.kind === 'write').length)} /><N.MetricTile label="Approve keys" value={String(capRows.filter((c) => c.kind === 'approve').length)} /><N.MetricTile label="Sensitive" value={String(capRows.filter((c) => c.sens).length)} /></div>
    <div style={{ maxWidth: 420 }}><N.TextField label="Search keys or screens" value={q} onChange={(e) => setQ(e.target.value)} placeholder="e.g. registrar.override" /></div>
    <N.FilterChips label="Group" options={GROUPS.map((G) => ({ id: G.g, label: G.g + ' · ' + G.label, count: capRows.filter((c) => c.g === G.g).length }))} selected={f} onToggle={(id) => setF((x) => x.includes(id) ? x.filter((y) => y !== id) : [...x, id])} onClear={() => setF([])} />
    <N.FilterChips label="Flags" options={[{ id: 'write', label: 'Write or approve' }, { id: 'high', label: 'High impact' }, { id: 'sens', label: 'Sensitive' }]} selected={flags} onToggle={(id) => setFlags((x) => x.includes(id) ? x.filter((y) => y !== id) : [...x, id])} onClear={() => setFlags([])} />
    <div className="scrollx"><N.DataTable caption={rows.length + ' keys'} columns={[{ key: 'key', label: 'Key', render: (c) => <code>{c.key}</code> }, { key: 'kind', label: 'Action' }, { key: 'n', label: 'Screen', render: (c) => <a href={'#screens/' + c.g + '/' + c.i}>{c.n}</a> }, { key: 'roles', label: 'Roles granted', numeric: true }, { key: 'flags', label: 'Flags', render: (c) => (c.high ? '! High ' : '') + (c.sens ? '◆ Sensitive' : '') }]} rows={rows.slice(0, lim)} /></div>
    {rows.length > lim && <div><N.Button onClick={() => setLim(lim + 300)}>Show 300 more ({rows.length - lim} left)</N.Button></div>}
  </div></div>;
}

}
