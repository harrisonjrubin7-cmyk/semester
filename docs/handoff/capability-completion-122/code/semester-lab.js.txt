(function(root,factory){const api=factory();if(typeof module==='object'&&module.exports)module.exports=api;else root.SemesterLab=api;})(typeof globalThis!=='undefined'?globalThis:this,function(){

const canonical=v=>JSON.stringify(sort(v));
function sort(v){if(Array.isArray(v))return v.map(sort);if(v&&typeof v==='object')return Object.fromEntries(Object.keys(v).sort().map(k=>[k,sort(v[k])]));return v;}
const validTime=x=>typeof x==='string'&&/T.*(?:Z|[+-]\d{2}:\d{2})$/.test(x)&&Number.isFinite(Date.parse(x));
function release(s,e,target,now=Date.now()){

 errors.push('local workbench has no verified server authority or professional/customer acceptance');

}
function check(cap,request,ctx,now=Date.now()){
 const checks=[];const add=(name,pass,reason)=>checks.push({name,pass:!!pass,reason:pass?'passed in synthetic context':reason});

 add('session assurance',c.sessionActive===true,'Active session required');

 add('workspace grant',Array.isArray(c.workspaces)&&c.workspaces.includes(cap.workspace),'Workspace grant missing');
 add('capability permission',Array.isArray(c.capabilities)&&c.capabilities.includes(cap.id),'Capability not granted');
 add('object scope',r.objectTenant===c.tenant&&r.objectOwner===c.actor,'Object scope denied');
 add('purpose/classification',typeof r.purpose==='string'&&!!r.purpose.trim()&&c.dataAllowed===true,'Purpose/data not allowed');

}
function createRunner(){const receipts=new Map();return function(cap,req,ctx,now){
 const decision=check(cap,req,ctx,now);if(!decision.allowed)return {kind:'blocked',decision};

 const receipt={id:'synthetic-'+(receipts.size+1),capability:cap.id,tenant:req.tenant,command:req.command,objectId:req.objectId,version:req.expectedVersion+1,label:'synthetic receipt; no external effect'};
 receipts.set(key,{binding:fingerprint,receipt});return {kind:'simulated',receipt,decision};
 };}
function buildOrder(caps){const map=new Map(caps.map(c=>[c.id,c])),seen=new Set(),visiting=new Set(),out=[];
 function visit(c){if(visiting.has(c.id))throw Error('Build dependency cycle: '+c.id);if(seen.has(c.id))return;visiting.add(c.id);for(const d of c.build_dependencies){if(!map.has(d))throw Error('Missing dependency: '+d);visit(map.get(d));}visiting.delete(c.id);seen.add(c.id);out.push(c.id);}caps.forEach(visit);return out;}
return {facts,states,canonical,check,release,createRunner,buildOrder};
