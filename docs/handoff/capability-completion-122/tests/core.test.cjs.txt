const test=require('node:test'),assert=require('node:assert/strict'),fs=require('node:fs'),path=require('node:path');
const L=require('../code/semester-lab.js');const caps=JSON.parse(fs.readFileSync(path.join(__dirname,'../registry/capabilities.json')));const now=Date.parse('2026-10-13T12:00:00Z');
function fixture(cap){const ctx={authenticated:true,actor:'demo-user',sessionActive:true,tenant:'synthetic-campus',workspaces:[cap.workspace],capabilities:[cap.id],dataAllowed:true,sourceAuthority:true,fresh:true,objectVersion:1,policyAllowed:true,consentAllowed:true,revoked:false,grantExpires:'2026-10-20T12:00:00Z'};const req={tenant:ctx.tenant,objectTenant:ctx.tenant,objectOwner:ctx.actor,objectId:'sample-object',purpose:'synthetic workflow test',expectedVersion:1,command:cap.commands[0],payload:{sample:true},key:'sample-key',mode:'consequential'};req.approval={approver:'demo-reviewer',tenant:req.tenant,capability:cap.id,command:req.command,objectId:req.objectId,expectedVersion:1,payloadBinding:L.canonical(req.payload),authority:'synthetic-reviewer',expires:'2026-10-20T12:00:00Z'};return {ctx,req};}
test('122 unique source capabilities',()=>{assert.equal(caps.length,122);assert.equal(new Set(caps.map(c=>c.id)).size,122);});
test('all specifications have bespoke goal/boundary and 11 acceptance checks',()=>{caps.forEach(c=>{assert.ok(c.goal&&c.boundary);assert.equal(c.acceptance.length,11);assert.equal(c.activation,'disabled');});});
test('build dependency graph is valid and ordered',()=>{const order=L.buildOrder(caps);assert.equal(order.length,122);caps.forEach(c=>c.build_dependencies.forEach(d=>assert.ok(order.indexOf(d)<order.indexOf(c.id))));});
test('dependency cycles rejected',()=>assert.throws(()=>L.buildOrder([{id:'a',build_dependencies:['b']},{id:'b',build_dependencies:['a']}])));
test('missing dependency rejected',()=>assert.throws(()=>L.buildOrder([{id:'a',build_dependencies:['b']}])));
for(const cap of caps){test(cap.id+' remains release-blocked without verified authority',()=>{assert.equal(L.release(cap,{},null,now).allowed,false);});if(!cap.commands.length)continue;
 test(cap.id+' authorized synthetic command / replay / changed payload',()=>{const {ctx,req}=fixture(cap),run=L.createRunner();assert.equal(run(cap,req,ctx,now).kind,'simulated');assert.equal(run(cap,req,ctx,now).kind,'replayed');req.payload={sample:false};req.approval.payloadBinding=L.canonical(req.payload);assert.equal(run(cap,req,ctx,now).kind,'blocked');});
 test(cap.id+' cross-tenant rejected',()=>{const {ctx,req}=fixture(cap);req.tenant='other';assert.equal(L.check(cap,req,ctx,now).allowed,false);});}
