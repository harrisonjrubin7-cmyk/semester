import {test} from 'node:test';import assert from 'node:assert/strict';import {spawn} from 'node:child_process';import {createInterface} from 'node:readline';import {mkdtemp,rm} from 'node:fs/promises';import {tmpdir} from 'node:os';import {join} from 'node:path';import {fileURLToPath} from 'node:url';import {createMockAssignmentsAdapter} from './mock-adapter.mjs';
const ctx={tenant:'a',actor:'student'};const input={area:'assignments',recordId:'x',version:'v1',actionId:'submit',fields:{text:'Synthetic work'}};
async function fixture(t){const d=await mkdtemp(join(tmpdir(),'semester-mock-'));const proc=spawn(process.env.PYTHON_EXECUTABLE||'python3',[fileURLToPath(new URL('./sqlite_worker.py',import.meta.url)),join(d,'mock.sqlite')],{stdio:['pipe','pipe','pipe']});const queue=[];let err='';proc.stderr.on('data',b=>err+=b);createInterface({input:proc.stdout}).on('line',line=>{const q=queue.shift();if(!q)return;const r=JSON.parse(line);r.error?q.reject(Error(r.error)):q.resolve(r.result)});proc.on('exit',()=>{while(queue.length)queue.shift().reject(Error('worker exited '+err))});const backend=p=>new Promise((resolve,reject)=>{queue.push({resolve,reject});proc.stdin.write(JSON.stringify(p)+'\n')});t.after(async()=>{proc.stdin.end();await new Promise(r=>proc.once('exit',r));await rm(d,{recursive:true,force:true})});return{backend,adapter:createMockAssignmentsAdapter(backend)}}
test('review writes nothing',async t=>{const f=await fixture(t);await f.adapter.review(ctx,input);assert.equal((await f.backend({op:'count'})).count,0)});
test('commit creates one submission',async t=>{const f=await fixture(t);assert.equal((await f.adapter.execute(ctx,input,'k')).status,'completed');assert.equal((await f.backend({op:'count'})).count,1)});
test('same key same payload returns same receipt',async t=>{const f=await fixture(t);const a=await f.adapter.execute(ctx,input,'k');const b=await f.adapter.execute(ctx,input,'k');assert.deepEqual(a,b);assert.equal((await f.backend({op:'count'})).count,1)});
test('conflicting key does not mutate',async t=>{const f=await fixture(t);await f.adapter.execute(ctx,input,'k');await assert.rejects(f.adapter.execute(ctx,{...input,fields:{text:'changed'}},'k'));assert.equal((await f.backend({op:'count'})).count,1)});
test('cross tenant denied',async t=>{const f=await fixture(t);await assert.rejects(f.adapter.review({...ctx,tenant:'b'},input));assert.equal((await f.backend({op:'count'})).count,0)});
test('wrong actor denied',async t=>{const f=await fixture(t);await assert.rejects(f.adapter.execute({...ctx,actor:'stranger'},input,'k'))});
test('revocation after review blocks commit',async t=>{const f=await fixture(t);await f.adapter.review(ctx,input);await f.backend({op:'grant',...ctx,record:'x',active:0});await assert.rejects(f.adapter.execute(ctx,input,'k'));assert.equal((await f.backend({op:'count'})).count,0)});
test('stale version denied before write',async t=>{const f=await fixture(t);await assert.rejects(f.adapter.execute(ctx,{...input,version:'old'},'k'));assert.equal((await f.backend({op:'count'})).count,0)});
test('invalid fields rejected',async t=>{const f=await fixture(t);await assert.rejects(f.adapter.review(ctx,{...input,fields:{text:''}}))});
test('unknown action rejected',async t=>{const f=await fixture(t);await assert.rejects(f.adapter.review(ctx,{...input,actionId:'grade'}))});
test('approval revoke blocks',async t=>{const f=await fixture(t);f.adapter.setApproval(false);await assert.rejects(f.adapter.review(ctx,input))});
test('read-only blocks write but permits lookup',async t=>{const f=await fixture(t);await f.adapter.execute(ctx,input,'k');f.adapter.setWrite(false);await assert.rejects(f.adapter.execute(ctx,input,'next'));assert.equal((await f.adapter.reconcile(ctx,input,'k')).status,'completed')});
test('timeout after durable commit reconciles without duplicate',async t=>{const f=await fixture(t);const port=async p=>{const r=await f.backend(p);if(p.op==='execute')throw Error('simulated connection loss AFTER commit');return r};const a=createMockAssignmentsAdapter(port);await assert.rejects(a.execute(ctx,input,'k'),/uncertain/);assert.equal((await a.reconcile(ctx,input,'k')).status,'completed');assert.equal((await f.backend({op:'count'})).count,1)});
test('unknown operation remains unresolved',async t=>{const f=await fixture(t);assert.equal(await f.adapter.reconcile(ctx,input,'unknown'),null)});
test('concurrent duplicate commands create one row',async t=>{const f=await fixture(t);await Promise.all([f.adapter.execute(ctx,input,'k'),f.adapter.execute(ctx,input,'k')]);assert.equal((await f.backend({op:'count'})).count,1)});
